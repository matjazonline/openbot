        heartbeat: Duration::from_secs(30),

    store: Arc<dyn LocalProtocolStore>,
    stop: CancellationToken,
    fence: ConnectionFence,
    deadline: Instant,
    receiver: oneshot::Receiver<WebSocket>,
    #[cfg(test)] command_write_gate: Option<Arc<tests::CommandWriteGate>>,
) {

    result=timeout_at(deadline,receiver)=>result.ok().and_then(Result::ok),};
    if let Some(mut socket) = socket {
        let initialized = tokio::select! {biased; _=stop.cancelled()=>Ok(false),
        result=timeout_at(deadline,driver::initialize(&mut socket,store.as_ref(),&stop,fence))=>
            result.unwrap_or_else(|_|Err(AppError::BadRequest("local machine handshake timed out".into()))),};
        match initialized {
            Ok(true) => {
                let result = tokio::select! {biased; _=stop.cancelled()=>Ok(()), result=driver::run(&mut socket,store.as_ref(),&stop,fence,
                #[cfg(test)] command_write_gate.as_deref())=>result};
                if result.is_err() {
                    record_issue(store.as_ref(), fence, ConnectionIssue::ConnectionFailed).await;
                    tracing::debug!(machine_id=%fence.machine.machine_id,"local machine connection ended");
                }
            }
            Ok(false) => {}
            Err(error) => {
                record_issue(store.as_ref(), fence, ConnectionIssue::ConnectionFailed).await;
                let kind = match error {
                    AppError::InvalidCredentials => "authentication",
                    AppError::BadRequest(_) => "protocol",
                    _ => "storage",
                };
                tracing::debug!(machine_id=%fence.machine.machine_id,error_kind=kind,"local machine handshake failed");
            }
        }
        let _ = timeout(Duration::from_secs(1), socket.close()).await;
    }
    if !matches!(
        timeout(
            Duration::from_secs(IO_SECONDS),
            store.release_connection(fence),
        )
        .await,
        Ok(Ok(()))
    ) {
        tracing::warn!(machine_id=%fence.machine.machine_id,"local connection lease cleanup failed");
    }
}

async fn heartbeat(
    store: Arc<dyn Store>,
    lease: Arc<Mutex<Lease>>,
    settings: IndexerSettings,
    stop: CancellationToken,
) {
    loop {
        tokio::select! {
            () = stop.cancelled() => return,
            () = tokio::time::sleep(settings.heartbeat) => {}
        }
        let held: Vec<IndexJob> = match lease.lock() {
            Ok(lease) => lease
                .held
                .iter()
                .filter(|job| !lease.lost.contains(&job.observation_id))
                .cloned()
                .collect(),
            Err(_) => return,
        };
        if held.is_empty() {
            continue;
        }
        let sent = Instant::now();
        // A failed renewal leaves the deadline where it was, so the work stops in time.
        if let Ok(renewed) = store.renew_index_leases(&held, settings.lease).await
            && let Ok(mut lease) = lease.lock()
        {
            lease.deadline = sent + settings.lease.saturating_sub(settings.margin);
            for job in held {
                if !renewed.contains(&job.observation_id) {
                    lease.lost.insert(job.observation_id);
                }
            }
        }
    }
}

export async function localCallerHeaders(origin, profile) {
  const url = new URL(origin);
  assert.equal(url.protocol, "http:");
  assert.equal(url.hostname, "127.0.0.1");
  const post = async (path, body) => {
    const response = await fetch(new URL(path, url), {
      method: "POST", headers: { "content-type": "application/json" },
      body: JSON.stringify(body), redirect: "error", signal: AbortSignal.timeout(5000),
    });
    assert(response.ok, "local caller proof refused");
    return response.json();
  };
  const started = await post("/api/v1/caller-proofs", {});
  assert.match(started.id, /^[0-9a-f]{12}-[0-9a-f]{32}$/);
  assert.equal(await realpath(started.directory), join(await realpath(profile), "caller-proofs"));
  const directory = await stat(started.directory);
  assert(directory.isDirectory() && directory.uid === process.getuid());
  const path = join(started.directory, started.id);
  await writeFile(path, started.id, { flag: "wx", mode: 0o600 });
  try {
    const { token } = await post(`/api/v1/caller-proofs/${started.id}`, { claim: started.claim });
    assert.match(token, /^[A-Za-z0-9_-]{32,128}$/);
    return { "x-cloudroom-caller": token };
  } finally {
    try { await unlink(path); } catch (error) { if (error.code !== "ENOENT") throw error; }
  }
}

pub fn authorize(
    grants: &[NamespaceGrant],
    operation: Operation,
    requested: &[String],
    read_ceiling: Option<AccessLevel>,
) -> Result<AuthorizedScope, Denial> {
    if requested.is_empty() {
        return Err(Denial::NoNamespace);
    }
    if requested.len() > MAX_NAMESPACES_PER_REQUEST {
        return Err(Denial::TooManyNamespaces);
    }
    if operation.reads() && read_ceiling.is_none() {
        return Err(Denial::MissingReadCeiling);
    }
    let unique: BTreeSet<&String> = requested.iter().collect();
    let mut namespace_ids = Vec::with_capacity(unique.len());
    let mut write_max: Option<AccessLevel> = None;
    for (index, name) in unique.into_iter().enumerate() {
        let grant = grants
            .iter()
            .find(|grant| &grant.namespace == name)
            .ok_or(Denial::NamespaceNotGranted)?;
        if !grant.operations.contains(&operation) {
            return Err(Denial::OperationNotGranted);
        }
        if let Some(ceiling) = read_ceiling {
            match grant.read_max {
                Some(read_max) if ceiling <= read_max => {}
                _ => return Err(Denial::CeilingExceedsGrant),
            }
        }
        write_max = if index == 0 {
            grant.write_max
        } else {
            match (write_max, grant.write_max) {
                (Some(left), Some(right)) => Some(left.min(right)),
                _ => None,
            }
        };
        if operation.writes() && grant.write_max.is_none() {
            return Err(Denial::WriteNotGranted);
        }
        namespace_ids.push(grant.namespace_id);
    }
    Ok(AuthorizedScope {
        namespace_ids,
        read_ceiling,
        write_max,
    })
}

async fn snapshot(
    db: &mut PgConnection,
    c: ConnectionIdentity<'_>,
) -> AppResult<AuthorizationSnapshot> {
    let grant = PostgresVault::grant_metadata_on(db, c.reference.grant).await?;
    let destination = ApprovedDestination::parse(c.destination)?;
    if !grant.enabled
        || grant.client != client_namespace()
        || grant.account != c.reference.account
        || grant.account_authorization != c.reference.authorization
        || grant.revision != c.reference.grant_revision
        || grant.binding != binding(c.company, c.id, c.kind)
        || grant.host_scope != company_binding(c.company)
        || grant.destination != destination
        || grant.purpose != c.kind.purpose()
    {
        return Err(missing());
    }
    Ok(AuthorizationSnapshot {
        account: grant.account,
        grant: grant.id,
        client: grant.client,
        binding: grant.binding,
        app: grant.app,
        app_policy: grant.app_policy,
        destination,
        account_authorization: grant.account_authorization,
        grant_revision: grant.revision,
        connection_revision: integration::ConnectionRevision::new(c.revision)?,
        catalog_revision: integration::CatalogRevision::new(c.catalog)?,
    })
}

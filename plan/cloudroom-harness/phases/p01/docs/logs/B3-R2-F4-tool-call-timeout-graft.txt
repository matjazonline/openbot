  requestId: string | number,
  method: string,
  params: unknown,
): DecodedToolCallRequest | null {
  if (method !== "item/tool/call") {
    return null;
  }

  const parsed = toolCallRequestParamsSchema.safeParse(params);
  if (!parsed.success) {
    return null;
  }

  return {
    requestId,
    providerThreadId: parsed.data.providerThreadId,
    turnId: parsed.data.turnId,
    callId: parsed.data.callId,
    ...(parsed.data.nativeInvocation ? {nativeInvocation:parsed.data.nativeInvocation} : {}),
    tool: parsed.data.tool,
    ...(parsed.data.arguments !== undefined
      ? { arguments: parsed.data.arguments }
      : {}),
    ...(parsed.data.threadId ? { threadId: parsed.data.threadId } : {}),
  };
}

export class NativeToolOrigins {
  private closed = false;
  private readonly pending: InputOwner[] = [];
  private readonly requests = new Set<string>();
  // Retain cancelled and acknowledged digests for this bounded SDK epoch too:
  // a delayed hook must never become proof for a later identical input.
  private readonly inputDigests = new Set<string>();
  private readonly ambiguousDigests = new Set<string>();
  private readonly prompts = new Map<string, InputOwner | null>();
  private readonly calls = new Map<string, ToolOwner | null>();

  enqueue(clientRequestId: string, input: string): string {
    if (this.closed || !validId(clientRequestId) || this.requests.has(clientRequestId)
      || this.pending.length >= NATIVE_ORIGIN_LIMITS.pending || this.requests.size >= NATIVE_ORIGIN_LIMITS.prompts) {
      throw new Error("Native input identity unavailable");
    }
    const owner = { token: randomUUID(), request: clientRequestId, digest: inputDigest(input) };
    this.requests.add(clientRequestId);
    if (this.inputDigests.has(owner.digest)) {
      this.ambiguousDigests.add(owner.digest);
      for (let i = this.pending.length - 1; i >= 0; i--) {
        if (this.pending[i]!.digest === owner.digest) this.pending.splice(i, 1);
      }
      return owner.token;
    }
    this.inputDigests.add(owner.digest);
    this.pending.push(owner);
    return owner.token;
  }

  cancel(token: string): void {
    const pending = this.pending.findIndex((owner) => owner.token === token);
    if (pending >= 0) this.pending.splice(pending, 1);
    for (const [prompt, owner] of this.prompts) if (owner?.token === token) this.prompts.set(prompt, null);
  }

  acknowledge(promptId: unknown, input: string, agentId?: string): boolean {
    if (this.closed || agentId !== undefined || !validId(promptId)) return false;
    const digest = inputDigest(input);
    if (this.prompts.has(promptId)) {
      const prior = this.prompts.get(promptId);
      if (prior?.digest === digest && !this.ambiguousDigests.has(digest)) return true;
      this.prompts.set(promptId, null);
      return false;
    }
    if (this.ambiguousDigests.has(digest)) return false;
    const owner = this.pending[0];
    if (!owner || owner.digest !== digest || this.prompts.size >= NATIVE_ORIGIN_LIMITS.prompts) return false;
    this.pending.shift();
    this.prompts.set(promptId, owner);
    return true;
  }

  observeTool(promptId: unknown, callId: unknown, tool: string, input: unknown, agentId?: string): boolean {
    if (this.closed || agentId !== undefined || !validId(promptId) || !validId(callId) || (!tool.startsWith(toolPrefix) || tool.length > 512)) return false;

      return false;
    }
    if (this.calls.size >= NATIVE_ORIGIN_LIMITS.calls) return false;
    this.calls.set(callId, { request: owner.request, prompt: promptId, tool, digest });

export function registerInternalToolCallRoutes(app: Hono, deps: AppDeps): void {
  const { post } = typedRoutes<HostDaemonInternalSchema>(app, {
    onValidationError: (msg) => new ApiError(400, "invalid_request", msg),
  });

  post(
    "/session/tool-call",
    hostDaemonToolCallRequestSchema,
    async (context, payload) => {
      const session = requireAuthenticatedDaemonSession({
        context,
        db: deps.db,
        sessionId: payload.sessionId,
      });
      const { environment, thread } = requireThreadEnvironment(
        deps.db,
        payload.threadId,
      );
      if (environment.hostId !== session.hostId) {
        throw new ApiError(
          403,
          "invalid_request",
          "Thread does not belong to the session host",
        );
      }

      if (payload.tool === UPDATE_ENVIRONMENT_DIRECTORY_TOOL_NAME) {
        return context.json(
          await handleUpdateEnvironmentDirectoryToolCall(deps, {
            currentEnvironment: environment,
            input: payload.arguments,
            thread,
            turnId: payload.turnId,
          }),
        );
      }

      const pluginTool = findPluginAgentTool(payload.tool);
      if (pluginTool) {
        const controller = new AbortController();
        const signal = AbortSignal.any([
          context.req.raw.signal,
          controller.signal,
        ]);
        return streamToolCallResponse(
          invokePluginAgentTool(pluginTool, {
            input: payload.arguments,
            ctx: {
              threadId: thread.id,
              projectId: thread.projectId,
              ...(payload.nativeInvocation ? {nativeInvocation:Object.freeze({...payload.nativeInvocation})} : {}),
              signal,
            },
          }),
          controller,
        );
      }

      return context.json({
        success: false,
        contentItems: [
          { type: "inputText", text: `Unsupported tool: ${payload.tool}` },
        ],
      });
    },
  );
}

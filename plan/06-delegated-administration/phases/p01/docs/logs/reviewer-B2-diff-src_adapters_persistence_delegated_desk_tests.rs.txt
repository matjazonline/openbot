+fn write(unit: DepartmentId, name: &str) -> ChannelWrite {
+    ChannelWrite {
+        name: name.into(),
+        slug: name.to_lowercase(),
+        department_id: Some(unit),
+        enabled: true,
+        ..Default::default()
+    }

+fn edit(d: &Delegated, actor: Uuid, desk: Uuid, unit: DepartmentId) -> ChannelUpdate {
+    ChannelUpdate {
+        actor: d.scope(actor).actor,
+        company_id: d.f.company,
+        channel_id: desk,
+        write: write(unit, "Edited"),
+    }
+}

+async fn state(d: &Delegated, desk: Uuid) -> (String, Uuid, i64) {

+    )
+    .bind(d.f.company)
+    .bind(desk)
+    .fetch_one(d.f.pool())
+    .await
+    .unwrap()

+#[tokio::test]
+async fn delegated_admin_desk_persistence_checks_current_destination_and_foreign_ownership() {
+    let Some(pool) = test_pool().await else {
+        return;
+    };
+    let d = Delegated::on(pool).await;
+    let remote = d.remote().await;
+    let p = &d.f.persistence;
+    let lead = d.scope(d.f.member).actor;
+    let sibling = d.scope(d.sibling_lead).actor;
+    let desk = ChannelPersistence::create(p, lead, d.f.company, write(remote.id, "Managed"))
+        .await
+        .unwrap();
+    let before = state(&d, desk.id).await;
+    let before_count: i64 = sqlx::query_scalar("SELECT count(*) FROM channels WHERE company_id=$1")
+        .bind(d.f.company)
+        .fetch_one(p.pool())
+        .await
+        .unwrap();
+    hidden(ChannelPersistence::create(p, sibling, d.f.company, write(remote.id, "Refused")).await);
+    hidden(ChannelPersistence::update(p, edit(&d, d.sibling_lead, desk.id, remote.id)).await);
+    hidden(ChannelPersistence::update(p, edit(&d, d.f.member, desk.id, d.sibling.id)).await);
+    let sibling_desk = d.f.desk_of(d.sibling.id).await;
+    hidden(ChannelPersistence::update(p, edit(&d, d.f.member, sibling_desk, remote.id)).await);
+    hidden(ChannelPersistence::delete(p, sibling, d.f.company, desk.id).await);
+    let foreign = crate::adapters::persistence::test_support::create_channel(
+        p,
+        d.f.foreign_company,
+        ChannelWrite {
+            name: "Foreign".into(),
+            slug: "foreign-target".into(),
+            ..Default::default()
+        },
+    )
+    .await
+    .unwrap();
+    hidden(ChannelPersistence::update(p, edit(&d, d.f.member, foreign.id, remote.id)).await);
+    hidden(ChannelPersistence::delete(p, lead, d.f.company, foreign.id).await);
+    hidden(
+        ChannelPersistence::create(
+            p,
+            lead,
+            d.f.company,
+            write(foreign.department_id, "Foreignowner"),
+        )
+        .await,
+    );
+    assert_eq!(state(&d, desk.id).await, before);
+    let after_count: i64 = sqlx::query_scalar("SELECT count(*) FROM channels WHERE company_id=$1")
+        .bind(d.f.company)
+        .fetch_one(p.pool())
+        .await
+        .unwrap();
+    assert_eq!(
+        after_count, before_count,
+        "refused creates leave no desk or partial row"
+    );
+    assert_eq!(
+        ChannelPersistence::get_by_id(p, sibling_desk)
+            .await
+            .unwrap()
+            .unwrap()
+            .department_id,
+        d.sibling.id
+    );
+    assert_eq!(
+        ChannelPersistence::get_by_id(p, foreign.id)
+            .await
+            .unwrap()
+            .unwrap()
+            .company_id,
+        d.f.foreign_company
+    );
+    ChannelPersistence::delete(p, lead, d.f.company, desk.id)
+        .await
+        .unwrap();
+}

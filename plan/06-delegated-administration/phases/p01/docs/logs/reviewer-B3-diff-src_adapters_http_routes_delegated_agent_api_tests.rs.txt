+#[tokio::test]
+async fn delegated_admin_agent_json_routes_allow_unit_managers_and_hide_sibling_and_foreign_targets()
+ {
+    let Some(database) = crate::adapters::persistence::test_support::own_database().await else {
+        return;
+    };
+    let f = Fixture::on(database.pool.clone()).await;
+    let (alice, unit) = lead(&f, "research").await;
+    let (bob, sibling) = lead(&f, "sales").await;
+    let own = created(&f, alice, unit, "delegated-json").await.unwrap();
+    let agent = Uuid::parse_str(own["agent"]["id"].as_str().unwrap()).unwrap();
+    assert_eq!(own["departments"][0]["id"], unit.as_uuid().to_string());
+    assert!(matches!(
+        created(&f, bob, unit, "sibling-json").await,
+        Err(crate::app_error::AppError::NotFound(_))
+    ));
+    assert!(matches!(
+        created(&f, alice, sibling, "outside-json").await,
+        Err(crate::app_error::AppError::NotFound(_))
+    ));
+    assert_eq!(
+        edit(&f, alice, agent).await.unwrap()["agent"]["name"],
+        "Edited by unit manager"
+    );
+    assert!(matches!(
+        edit(&f, bob, agent).await,
+        Err(crate::app_error::AppError::NotFound(_))
+    ));
+    let departments =
+        crate::adapters::persistence::test_support::department_use_cases(&f.persistence);
+    assert!(matches!(
+        get_agent_json(
+            State(f.agents.clone()),
+            State(f.skills.clone()),
+            State(departments.clone()),
+            AuthenticatedUser { id: bob },
+            Path((f.company, agent))
+        )
+        .await,
+        Err(crate::app_error::AppError::NotFound(_))
+    ));
+    let foreign = Fixture::on(database.pool.clone()).await;
+    assert!(matches!(
+        get_agent_json(
+            State(f.agents.clone()),
+            State(f.skills.clone()),
+            State(departments),
+            AuthenticatedUser { id: alice },
+            Path((foreign.company, agent))
+        )
+        .await,
+        Err(crate::app_error::AppError::NotFound(_))
+    ));
+    assert!(matches!(
+        delete_agent_json(
+            State(f.agents.clone()),
+            AuthenticatedUser { id: bob },
+            Path((f.company, agent))
+        )
+        .await,
+        Err(crate::app_error::AppError::NotFound(_))
+    ));
+    assert!(matches!(
+        delete_agent_json(
+            State(f.agents.clone()),
+            AuthenticatedUser { id: alice },
+            Path((foreign.company, agent))
+        )
+        .await,
+        Err(crate::app_error::AppError::NotFound(_))
+    ));
+    delete_agent_json(
+        State(f.agents.clone()),
+        AuthenticatedUser { id: alice },
+        Path((f.company, agent)),
+    )
+    .await
+    .unwrap();
+    assert!(
+        AgentPersistence::get_by_id(f.persistence.as_ref(), agent)
+            .await
+            .unwrap()
+            .is_none()
+    );
+}

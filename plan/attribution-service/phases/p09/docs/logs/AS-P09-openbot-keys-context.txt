async fn memory_service_startup(
    foundation: &foundation::Foundation,
) -> anyhow::Result<Option<crate::infra::memory_config::MemoryServiceConfig>> {
    let config =
        crate::infra::memory_config::MemoryServiceConfig::from_env().map_err(anyhow::Error::msg)?;
    let pool = foundation.persistence.pool();
    match &config {
        Some(config) => {
            tracing::info!(
                provisioner_credential_id = %config.provisioner.credential_id(),
                interactive_in_flight = config.interactive_in_flight,
                background_in_flight = config.background_in_flight,
                "Memory service configured; not contacted at startup"
            );
            // Failed lifecycles need an operator (docs/deploy.md, "Memory service").
            let (failed, more) =
                crate::adapters::persistence::memory::pending_lifecycles(pool, &["failed"], 20)
                    .await?;
            if !failed.is_empty() {
                tracing::warn!(
                    lifecycles = ?failed,
                    more,
                    "Memory lifecycles failed (not retried until re-driven) or stuck in an outage"
                );
            }
        }
        None => {
            let (pending, more) = crate::adapters::persistence::memory::pending_lifecycles(
                pool,
                &["pending", "claimed", "failed"],
                20,
            )
            .await?;
            if !pending.is_empty() {
                tracing::warn!(
                    lifecycles = ?pending,
                    more,
                    "MEMORY_SERVICE_URL is unset while memory lifecycles are unfinished; their \
                     remote tenants and namespaces will not be reconciled"
                );
            }
        }
    }
    Ok(config)
}

pub(super) async fn build(config: Arc<AppConfig>) -> anyhow::Result<Foundation> {
    let model_path = config.knowledge_model_dir.clone();
    // Load exactly one local model before database setup or worker construction.
    let mut loading = tokio::task::spawn_blocking(move || {
        crate::adapters::embedding::LocalTextEmbedder::load(&model_path)
    });
    let embedder =
        match tokio::time::timeout(std::time::Duration::from_secs(60), &mut loading).await {
            Ok(result) => result??,
            Err(_) => {
                // Native calls cannot be interrupted; Serve's runtime shutdown bound is the backstop.
                loading.abort();
                anyhow::bail!("Knowledge model startup exceeded sixty seconds");
            }
        };
    let smtp =
        LettreMailTransport::from_config(&config, smtp_allow_plaintext_local_from_env()).await?;
    let mail_available = smtp.is_configured();
    let mail_transport: Arc<dyn MailTransport> = smtp;
    let monitoring: Arc<dyn MonitoringService> = Arc::new(CompositeMonitor::new(vec![
        Arc::new(TracingMonitor::new()),
        Arc::new(InMemoryMonitor::new()),
    ]));
    // Validate configured storage before slower database work, as the original composition did.
    let storage = match config.gcs.as_ref() {
        Some(gcs) => Some(Arc::new(GcsFileStorage::from_config(gcs)?) as Arc<dyn FileStorage>),
        None => None,
    };
    let persistence = Arc::new(postgres_persistence().await?);
    let knowledge = Arc::new(crate::application::knowledge::KnowledgeLibrary::new(
        persistence.clone(),
        Arc::new(embedder),
    ));
    let sessions = Arc::new(
        SessionAuthority::new(&config, persistence.clone()).with_monitoring(monitoring.clone()),
    );
    let attribution = crate::infra::attribution::build(persistence.clone(), sessions.clone())?;
    install_builtin_agent_library(persistence.as_ref()).await?;
    Ok(Foundation {
        attribution,
        knowledge,
        config,
        persistence,
        monitoring,
        storage,
        mail_transport,
        mail_available,
        sessions,
    })
}

    pub fn from_lookup(lookup: impl Fn(&str) -> Option<String>) -> anyhow::Result<Option<Self>> {
        let names = [
            "OPENBOT_ATTRIBUTION_ORIGIN",
            "OPENBOT_ATTRIBUTION_CLIENT_SECRET",
            "OPENBOT_ATTRIBUTION_LIFECYCLE_SECRET",
            "OPENBOT_ATTRIBUTION_PREVIOUS_CLIENT_SECRET",
            "OPENBOT_ATTRIBUTION_PREVIOUS_CLIENT_SECRET_EXPIRES_AT",
        ];
        if names.iter().all(|name| lookup(name).is_none()) {
            return Ok(None);
        }
        let required =
            |name: &str| lookup(name).ok_or_else(|| anyhow::anyhow!("{name} is required"));
        let raw = required(names[0])?;
        if raw.len() > 2048 {
            anyhow::bail!("OPENBOT_ATTRIBUTION_ORIGIN is too long");
        }
        let origin = Url::parse(&raw)
            .map_err(|_| anyhow::anyhow!("OPENBOT_ATTRIBUTION_ORIGIN is invalid"))?;
        let loopback = match origin.host() {
            Some(url::Host::Domain("localhost")) => true,
            Some(url::Host::Ipv4(ip)) => ip.is_loopback(),
            Some(url::Host::Ipv6(ip)) => ip.is_loopback(),
            _ => false,
        };
        if origin.host().is_none()
            || !(origin.scheme() == "https" || (origin.scheme() == "http" && loopback))
            || !origin.username().is_empty()
            || origin.password().is_some()
            || origin.path() != "/"
            || origin.query().is_some()
            || origin.fragment().is_some()
        {
            anyhow::bail!(
                "OPENBOT_ATTRIBUTION_ORIGIN must be an HTTPS origin (loopback HTTP for tests), without credentials/path/query/fragment"
            );
        }
        let client = required(names[1])?;
        let cursor_key = secret(&client)?;
        let lifecycle_secret = required(names[2])?;
        secret(&lifecycle_secret)?;
        if lifecycle_secret == client || lookup(names[3]).as_ref() == Some(&lifecycle_secret) {
            anyhow::bail!("attribution client and lifecycle credentials must be distinct");
        }
        let previous = match (lookup(names[3]), lookup(names[4])) {
            (None, None) => None,
            (Some(value), Some(expiry)) => {
                secret(&value)?;
                let expiry = DateTime::parse_from_rfc3339(&expiry)
                    .map_err(|_| {
                        anyhow::anyhow!("previous attribution secret expiry must be RFC3339")
                    })?
                    .with_timezone(&Utc);
                if expiry <= Utc::now() || expiry > Utc::now() + chrono::Duration::days(1) {
                    anyhow::bail!("previous attribution secret expiry must be within 24 hours");
                }
                Some((hash(&value).0, expiry))
            }
            _ => anyhow::bail!("previous attribution client secret requires its expiry"),
        };
        Ok(Some(Self {
            callback: format!("{origin}auth/callback"),
            delivery: format!("{origin}internal/v1/company-lifecycle"),
            origin,
            client_hash: hash(&client).0,
            previous,
            cursor_key,
            lifecycle_secret,
        }))
    }

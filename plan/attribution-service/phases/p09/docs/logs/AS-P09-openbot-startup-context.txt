pub(super) async fn build(config: Arc<AppConfig>) -> anyhow::Result<Foundation> {
    let model_path = config.knowledge_model_dir.clone();
    // Load exactly one local model before database setup or worker construction.
    let mut loading = tokio::task::spawn_blocking(move || {
        crate::adapters::embedding::LocalTextEmbedder::load(&model_path)
    });
    let embedder =
        match tokio::time::timeout(std::time::Duration::from_secs(60), &mut loading).await {
            Ok(result) => result??,
            Err(_) => {
                // Native calls cannot be interrupted; Serve's runtime shutdown bound is the backstop.
                loading.abort();
                anyhow::bail!("Knowledge model startup exceeded sixty seconds");
            }
        };
    let smtp =
        LettreMailTransport::from_config(&config, smtp_allow_plaintext_local_from_env()).await?;
    let mail_available = smtp.is_configured();
    let mail_transport: Arc<dyn MailTransport> = smtp;
    let monitoring: Arc<dyn MonitoringService> = Arc::new(CompositeMonitor::new(vec![
        Arc::new(TracingMonitor::new()),
        Arc::new(InMemoryMonitor::new()),
    ]));
    // Validate configured storage before slower database work, as the original composition did.
    let storage = match config.gcs.as_ref() {
        Some(gcs) => Some(Arc::new(GcsFileStorage::from_config(gcs)?) as Arc<dyn FileStorage>),
        None => None,
    };
    let persistence = Arc::new(postgres_persistence().await?);
    let knowledge = Arc::new(crate::application::knowledge::KnowledgeLibrary::new(
        persistence.clone(),
        Arc::new(embedder),
    ));
    let sessions = Arc::new(
        SessionAuthority::new(&config, persistence.clone()).with_monitoring(monitoring.clone()),
    );
    let attribution = crate::infra::attribution::build(persistence.clone(), sessions.clone())?;
    install_builtin_agent_library(persistence.as_ref()).await?;
    Ok(Foundation {
        attribution,
        knowledge,
        config,
        persistence,
        monitoring,
        storage,
        mail_transport,
        mail_available,
        sessions,
    })
}

pub(super) fn workflow(
    f: &foundation::Foundation,
    clients: &clients::Clients,
    mail: &Mail,
    memory: Option<&crate::infra::memory_config::MemoryServiceConfig>,
) -> anyhow::Result<Arc<crate::infra::workflow::WorkflowEngine>> {
    let engine = crate::infra::workflow::WorkflowEngine::new(
        (*f.persistence).clone(),
        crate::infra::workflow::WorkflowEngineSetup {
            integrations: clients.integrations.clone(),
            config: &f.config,
            composer: mail.composer.clone(),
            mcp_client: clients.mcp.clone(),
            monitoring: f.monitoring.clone(),
            slots: crate::infra::workflow::worker_slots(
                std::env::var("WORKFLOW_WORKER_SLOTS").ok().as_deref(),
                crate::infra::db::max_connections(),
            )?,
            knowledge: f.knowledge.clone(),
            // No company HTTP connection may reach the memory service (phase 05 E5).
            http_destinations: match memory {
                Some(config) => crate::adapters::workflow_http::HttpDestinationPolicy::default()
                    .reserving(&config.base_url)?,
                None => crate::adapters::workflow_http::HttpDestinationPolicy::default(),
            },
        },
    )?
    .with_jev_assets(Arc::new(
        crate::adapters::workflow_jev::JevAssetReader::new(
            (*f.persistence).clone(),
            f.storage.clone(),
        ),
    ));
    let engine = match memory {
        Some(config) => {
            let service = Arc::new(crate::adapters::memory_service::MemoryServiceHttp::new(
                config,
            )?);
            engine.with_memory(
                service.clone(),
                service,
                config.provisioner.credential_id(),
                config.background_in_flight,
            )
        }
        None => engine,
    };
    Ok(Arc::new(engine))
}

pub(super) fn build(
    f: &foundation::Foundation,
    memory: Option<&crate::infra::memory_config::MemoryServiceConfig>,
) -> anyhow::Result<Clients> {
    let mut policy = crate::adapters::mcp::policy::EndpointPolicy::new(
        std::env::var("MCP_INTERNAL_ENDPOINTS")
            .unwrap_or_default()
            .split(',')
            .map(str::trim)
            .filter(|value| !value.is_empty())
            .map(str::to_owned),
    )?;
    // No company MCP connection may reach the memory service (phase 05 E5).
    if let Some(memory) = memory {
        policy = policy.reserving(&memory.base_url)?;
    }
    let mcp = Arc::new(crate::adapters::mcp::HttpMcpClient::new(policy));
    let p = &f.persistence;
    let services = crate::infra::integrations::gateway(
        p,
        &f.config,
        mcp.clone(),
        memory.map(|memory| memory.base_url.as_str()),
    )?;
    let mcp_use_cases = Arc::new(crate::use_cases::mcp::McpUseCases::new(
        p.clone(),
        p.clone(),
        p.clone(),
        mcp.clone(),
        services.discovery,
    ));
    let resend = Arc::new(CompanyResendApiClients::new(
        p.clone(),
        f.config.resend_api.clone(),
    )?);
    Ok(Clients {
        integrations: services.gateway,
        integration_bindings: services.bindings,
        mcp,
        mcp_use_cases,
        resend,
    })
}

pub(super) fn build(f: &foundation::Foundation) -> anyhow::Result<Mail> {
    let email = Arc::new(EmailRenderer::new(&f.config.app_domain_name));
    let renderers = Arc::new(
        TransportRenderers::new()
            .register(email.clone())
            .map_err(|error| anyhow::anyhow!("Could not register the email renderer: {error}"))?,
    );
    let composer = DeliveryComposer::new(renderers.clone(), f.persistence.clone());
    Ok(Mail {
        email,
        renderers,
        composer,
    })
}

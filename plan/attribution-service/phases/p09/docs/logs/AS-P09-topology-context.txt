        .await

    assert_eq!(login.status(), StatusCode::FOUND);
    let cookie = login.headers()["set-cookie"]
        .to_str()
        .unwrap()
        .split(';')
        .next()
        .unwrap()
        .to_owned();
    let location = url::Url::parse(login.headers()["location"].to_str().unwrap()).unwrap();
    let state = location
        .query_pairs()
        .find(|(key, _)| key == "state")
        .unwrap()
        .1
        .into_owned();
    for (state, cookie) in [
        (state.clone(), None),
        (crypto::random(), Some(cookie.clone())),
        (
            state,
            Some(format!("__Host-attribution-login={}", crypto::random())),
        ),
    ] {
        let mut request = Request::builder().uri(format!(
            "/auth/callback?code={}&state={state}",
            crypto::random()
        ));
        if let Some(cookie) = cookie {
            request = request.header("cookie", cookie);
        }
        let response = f
            .app
            .clone()
            .oneshot(request.body(Body::empty()).unwrap())
            .await
            .unwrap();
        assert_eq!(response.status(), StatusCode::UNAUTHORIZED);
        assert!(!response.headers().contains_key("set-cookie"));
    }
    assert_eq!(f.upstream.calls.load(Ordering::Acquire), 0);
    let mut limited = false;
    for _ in 0..40 {
        let response = f
            .app
            .clone()
            .oneshot(
                Request::builder()
                    .uri("/auth/login")
                    .body(Body::empty())
                    .unwrap(),
            )
            .await

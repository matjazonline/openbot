async fn agent_write(
    tx: &mut Transaction<'_, Postgres>,
    company: CompanyId,
    spec: &CreateAgentChannel,
    created_by: &CreationProvenance,
) -> AppResult<Provisioned<AgentWrite>> {
    let ready: Vec<_> = ready_models_on(tx, company)
        .await?
        .into_iter()
        .map(
            |(provider, model)| crate::entities::company::CompanyModelConnection {
                provider: provider.into(),
                models: vec![model.into()],
                is_default: false,
                has_api_key: true,
            },
        )
        .collect();
    if !model_allowed(&ready, &spec.provider, &spec.model) {
        return Ok(Err(ProvisioningRefusal::Invalid(format!(
            "Model '{}' of provider '{}' is not enabled with a key in this company",
            spec.model, spec.provider
        ))));
    }
    let mut tools = Vec::with_capacity(spec.tools.len());
    for name in &spec.tools {
        let Ok(name) = TypeName::parse(name) else {
            return Ok(Err(ProvisioningRefusal::Invalid(format!(
                "'{name}' is not a tool name"
            ))));
        };
        if is_builtin_provisioning_tool(name.as_str()) {
            return Ok(Err(ProvisioningRefusal::Invalid(format!(
                "{name} cannot be granted to a created agent"
            ))));
        }
        if !tools.contains(&name) {
            tools.push(name);
        }
    }
    let mut skill_ids = Vec::with_capacity(spec.skills.len());
    for slug in &spec.skills {
        let id: Option<Uuid> = sqlx::query_scalar(
            "SELECT id FROM skills WHERE company_id = $1 AND slug = $2 FOR KEY SHARE",
        )
        .bind(company.as_uuid())
        .bind(slug)
        .fetch_optional(&mut **tx)
        .await?;
        match id {
            Some(id) if !skill_ids.contains(&id) => skill_ids.push(id),
            Some(_) => {}
            None => {
                return Ok(Err(ProvisioningRefusal::Invalid(format!(
                    "No skill named '{slug}' exists in this company"
                ))));
            }
        }
    }
    Ok(Ok(AgentWrite {
        name: spec.name.clone(),
        slug: spec.slug.clone(),
        provider: Some(spec.provider.clone()),
        model: Some(spec.model.clone()),
        system_prompt: Some(spec.system_prompt.clone()),
        description: spec.description.clone(),
        tool_bindings: Some(
            super::super::agent_binding_selection::resolve_bindings_on(
                tx,
                company.as_uuid(),
                None,
                &tools,
            )
            .await?,
        ),
        skill_ids,
        created_by: Some(created_by.clone()),
        ..AgentWrite::default()
    }))
}

    pub(crate) fn normalize(&mut self) -> AppResult<()> {
        self.name = self.name.trim().to_string();
        self.slug = self.slug.trim().to_lowercase().replace(' ', "-");

        if self.name.is_empty() || self.slug.is_empty() {
            return Err(AppError::BadRequest(
                "The agent name and slug cannot be empty.".into(),
            ));
        }
        validate_slug(&self.slug, SlugKind::AgentSlug)?;

        for field in [
            &mut self.provider,
            &mut self.model,
            &mut self.system_prompt,
            &mut self.description,
        ] {
            if let Some(value) = field.as_mut() {
                *value = value.trim().to_string();
                if value.is_empty() {
                    *field = None;
                }
            }
        }

        if let Some(bindings) = &self.tool_bindings {
            crate::entities::agent_capability::validate_agent_bindings(bindings)
                .map_err(AppError::BadRequest)?;
        }
        deduplicate(&mut self.skill_ids);
        if self.skill_ids.len() > MAX_AGENT_SKILLS {
            return Err(AppError::BadRequest(format!(
                "An agent may carry at most {MAX_AGENT_SKILLS} skills."
            )));
        }

        Ok(())
    }

pub fn validate_agent_bindings(bindings: &[AgentToolBinding]) -> Result<(), String> {
    if bindings.len() > MAX_AGENT_BINDINGS {
        return Err("An agent may have at most 32 tool bindings".into());
    }
    let mut aliases = BTreeSet::new();
    let mut ids = BTreeSet::new();
    for binding in bindings {
        binding.validate()?;
        if !aliases.insert(&binding.alias) || !ids.insert(binding.id) {
            return Err("Agent tool aliases and binding identities must be distinct".into());
        }
    }
    Ok(())
}

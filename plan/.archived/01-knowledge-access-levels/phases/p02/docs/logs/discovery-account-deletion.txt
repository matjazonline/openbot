            .unwrap()

pub(crate) async fn rekey_account_identity_on(
    connection: &mut sqlx::PgConnection,
    user_id: Uuid,
    old_email: &str,
    new_email: &str,
) -> AppResult<()> {
    let persons: Vec<PersonPrincipal> = sqlx::query_as(
        r#"SELECT company_id, id FROM principals
           WHERE user_id = $1 AND kind = 'person'
           ORDER BY company_id"#,
    )
    .bind(user_id)
    .fetch_all(&mut *connection)
    .await
    .map_err(AppError::from)?;
    if persons.is_empty() {
        return Ok(());
    }
    // Registration accepts any plausible address, so the old one may be one no identity was ever
    // written for; there is then nothing of it to detach. The new one becomes an identity, so it
    // must parse, exactly as it must when a company is created.
    let old = account_email_identity(old_email).ok();
    let new = account_email_identity(new_email)?;
    if old.as_ref() == Some(&new) {
        return Ok(());
    }
    let username: String = sqlx::query_scalar("SELECT username::text FROM users WHERE id = $1")
        .bind(user_id)
        .fetch_one(&mut *connection)
        .await
        .map_err(AppError::from)?;

    // Both identity locks are taken in one order everywhere, so two re-keys that touch the same
    // pair of addresses cannot wait on each other crosswise.
    let mut keys: Vec<&QualifiedIdentity> = old.iter().chain([&new]).collect();
    keys.sort_by(|left, right| left.subject().as_str().cmp(right.subject().as_str()));

    for person in persons {
        for key in &keys {
            lock_identity_key(connection, person.company_id, key).await?;
        }
        let principal = PrincipalId::new(person.id);
        if let Some(old) = &old {
            detach_old_address(connection, person.company_id, principal, old).await?;
        }
        adopt_new_address(connection, person.company_id, principal, &new, &username).await?;
    }
    Ok(())
}

async fn account_email_change_old_address_is_adopted_by_its_next_owner() {
    let Some(pool) = test_pool().await else {
        return;
    };
    let persistence = PostgresPersistence::new(pool);
    let user = verified_account(&persistence, "leaver").await;
    let company_id = company_of(&persistence, &user, "adopt").await;
    let old = user.email.clone();
    let new = format!("{}@example.test", unique("left"));
    change_address(&persistence, &user, &new, AddressChange::Confirmed).await;
    let outsider = access(&persistence, company_id, &old)
        .await
        .principal_id
        .unwrap();

    let successor = persistence
        .create_user(&unique("successor"), &old, "hash")
        .await
        .unwrap();
    sqlx::query("UPDATE users SET email_verified_at = CURRENT_TIMESTAMP WHERE id = $1")
        .bind(successor.id)
        .execute(persistence.pool())
        .await
        .unwrap();
    let successor = UserPersistence::get_by_id(&persistence, successor.id)
        .await
        .unwrap()
        .unwrap();
    join(&persistence, company_id, &successor).await;

    let adopted = access(&persistence, company_id, &old).await;
    assert_eq!(
        adopted.principal_id,
        Some(outsider),
        "the outsider is adopted, not duplicated"
    );
    assert_eq!(adopted.membership, CompanyMembership::Member);
    assert_eq!(
        principal_kind(&persistence, outsider.as_uuid()).await,
        "person"
    );
    // The first account still holds the company through its new address.
    assert_eq!(
        access(&persistence, company_id, &new).await.membership,
        CompanyMembership::Owner
    );

    let _ = CompanyPersistence::delete(&persistence, company_id).await;
}

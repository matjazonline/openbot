import hashlib,json,subprocess,sys,os,stat
from pathlib import Path
exclude="plan/01-knowledge-access-levels/phases/p04/docs/"
paths=set()
for command in [["git","ls-files","-z"],["git","ls-files","--others","--exclude-standard","-z"]]:
 paths.update(x.decode()for x in subprocess.check_output(command).split(b"\0")if x)
rows=[]
for name in sorted(paths):
 if name.startswith(exclude) or name==exclude.rstrip("/"):continue
 p=Path(name);info=p.lstat()
 if stat.S_ISLNK(info.st_mode):kind="symlink";data=os.fsencode(os.readlink(p))
 elif stat.S_ISREG(info.st_mode):kind="file";data=p.read_bytes()
 else:raise AssertionError(name)
 rows.append({"path":name,"kind":kind,"executable":bool(info.st_mode&0o111),"size":len(data),"sha256":hashlib.sha256(data).hexdigest()})
canonical=json.dumps(rows,sort_keys=True,separators=(",",":")).encode()
result={"head":subprocess.check_output(["git","rev-parse","HEAD"],text=True).strip(),"exclusions":[exclude.rstrip("/")],"files_count":len(rows),"tree_sha256":hashlib.sha256(canonical).hexdigest(),"files":rows}
Path(sys.argv[1]).write_text(json.dumps(result,indent=2)+"\n")
print({k:v for k,v in result.items()if k!="files"})

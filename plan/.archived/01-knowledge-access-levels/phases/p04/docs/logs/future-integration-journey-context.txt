        LlmTurn::text("\"We are on it.\""),
        LlmTurn::text("\"We are on it.\""),

    let binding = d
        .bind(answering_source(), Some(MessageTrigger::Direct))
        .await;

    let second = d.email(customer_email(&d, "second", Some("first"))).await;

        second.thread.as_ref().unwrap().id,

        "one conversation"
    );

    d.run_until("two replies", async || replies(&d).await.len() == 2)
        .await;

        .iter()

            ingest
                .inbound_message
                .as_ref()
                .unwrap()
                .canonical_id
                .as_uuid()
        })
        .collect();
    let runs: Vec<(Uuid, Uuid, Value)> = sqlx::query_as(
        "SELECT id, trigger_id, input FROM workflow_runs WHERE binding_id = $1 ORDER BY created_at",
    )

    .await

    let replies = replies(&d).await;
    for (index, ingest) in [&first, &second].iter().enumerate() {

        assert_eq!(answered.len(), 1, "one reply per run");
        assert_eq!((answered[0].1, answered[0].2), (sources[index], thread));
    }
    assert_eq!(
        d.count("SELECT count(*) FROM workflow_message_intents WHERE company_id = $1 AND outcome = 'admitted'")
            .await,
        2
    );
    assert_eq!(d.llm.finish().await.unwrap(), 2);

fn an_agent_reply_is_drawn_as_its_channel_address_and_an_inbound_message_as_its_sender() {
    let company = mailbox_company();
    let thread = mailbox_thread(mailbox_channel(company.id).id);
    let scope = MessageScope {
        company_id: Uuid::new_v4(),
        channel_id: Uuid::new_v4(),
    };

    // A reply is stored with the channel's address as its sender, and that is what it shows.
    let reply = stored_email_view(EmailMessageDraft {
        role: MessageRole::Agent,
        direction: MessageDirection::Outbound,
        sender: EmailAddress::from("support@acme.mailagents.com"),
        ..mailbox_message_draft(thread.id, "The reply.")
    });
    let html = message_bubble_chat(&reply, None, scope, RunLinkAccess::Label);
    assert!(!html.contains("<img"));
    assert!(html.contains("support@acme.mailagents.com"));

    // An inbound message is its sender's.
    let inbound = mailbox_message(thread.id, "The question.");
    let inbound_html = message_bubble_chat(&inbound, None, scope, RunLinkAccess::Label);
    assert!(!inbound_html.contains("<img"));

    assert!(html.contains(r#"data-role="agent""#));

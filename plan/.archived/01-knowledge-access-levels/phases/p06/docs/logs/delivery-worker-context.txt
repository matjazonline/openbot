async fn a_delivery_for_an_unregistered_transport_is_dead_lettered() {
    let queue = Arc::new(ScriptedQueue::default());
    let worker = DeliveryWorker::new(queue.clone(), Arc::new(TransportRegistry::new()));
    let delivery = claimed(
        uuid::Uuid::new_v4(),
        vec![stored_part(0, DeliveryPartStatus::Prepared)],
    );

    let failures = queue.failures.lock().unwrap();

    assert_eq!(failures[0].0, FailureClass::Internal);
    assert_eq!(
        failures[0].1,
        Disposition::Terminal,
        "a missing adapter cannot come out differently next time"
    );
    assert!(
        queue.started.lock().unwrap().is_empty(),
        "nothing may be marked as started when there is nothing to send it"
    );
}

        return;
    };
    let persistence = PostgresPersistence::new(database.pool.clone());

    let mut deliveries = Vec::new();
    for index in 0..3 {
        let queued = queue(
            &persistence,
            &scope,
            DeliveryFixtureRequest::new(
                scope.company.id,
                scope.channel.id,
                scope.thread_id,
                &format!("slow-{index}"),
            ),
        )
        .await;
        deliveries.push(queued.delivery.id);
    }
    let transports = TransportRegistry::new()
        .register(
            Arc::new(EmailRenderer::new("example.test")),
            Arc::new(SlowProvider {
                delay: Duration::from_millis(1200),
            }),
        )
        .unwrap();
    let worker = DeliveryWorker::new(Arc::new(persistence.clone()), Arc::new(transports))
        .with_lease(Duration::from_secs(2));

    let claimed = worker.drain_once(&receiver).await.unwrap();

    assert_eq!(claimed, 3);
    for delivery in deliveries {
        assert_eq!(
            status_of(&persistence, delivery).await,
            DeliveryStatus::Delivered,
            "every delivery of the batch was sent, the last one included"
        );
    }
}

        .execute(&fixture.pool)
        .await;
    assert!(unknown_kind.is_err());
    fixture.cleanup().await;

async fn the_stuck_delivery_census_counts_each_stall_and_not_a_due_delivery() {
    let Some(database) = own_database().await else {
        return;
    };
    let persistence = PostgresPersistence::new(database.pool.clone());
    let scope = scope(&persistence).await;
    let mut ids = Vec::new();
    for key in [
        "census-dead",
        "census-overdue",
        "census-unknown",
        "census-due",
        "census-sent",
    ] {
        let fixture = queue(
            &persistence,
            &scope,
            DeliveryFixtureRequest::new(scope.company.id, scope.channel.id, scope.thread_id, key),
        )
        .await;
        ids.push(fixture.delivery.id.as_uuid());
    }
    let set = |id: Uuid, status: &'static str, available: chrono::Duration| {
        let pool = database.pool.clone();
        async move {
            sqlx::query(
                "UPDATE message_deliveries
                    SET status = $2, available_at = CURRENT_TIMESTAMP - $3::interval,
                        delivered_at = CASE WHEN $2 = 'delivered' THEN CURRENT_TIMESTAMP END,
                        last_error_class = CASE WHEN $2 IN ('retryable','dead_letter') THEN 'invalid_payload' END
                  WHERE id = $1",
            )
            .bind(id)
            .bind(status)
            .bind(sqlx::postgres::types::PgInterval::try_from(available.to_std().unwrap()).unwrap())
            .execute(&pool)
            .await
            .unwrap();
        }
    };
    set(ids[0], "dead_letter", chrono::Duration::hours(1)).await;
    set(ids[1], "retryable", chrono::Duration::minutes(10)).await;
    set(ids[2], "outcome_unknown", chrono::Duration::hours(1)).await;
    // Due, but inside the threshold: the worker has not fallen behind on it yet.
    set(ids[3], "pending", chrono::Duration::minutes(1)).await;
    set(ids[4], "delivered", chrono::Duration::hours(1)).await;

    let census = persistence
        .census_stuck_deliveries(Duration::from_secs(5 * 60))
        .await
        .unwrap();

    assert_eq!(
        census,
        crate::entities::stuck_work::StuckWorkCensus {
            delivery_dead_lettered: 1,
            delivery_overdue: 1,
            delivery_unconfirmed: 1,
        }
    );
}

async fn notification_delivery_tenant_part_company_must_match_its_delivery() {
    let Some(pool) = test_pool().await else {
        return;
    };
    let persistence = PostgresPersistence::new(pool);
    let home = scope(&persistence).await;
    let foreign = scope(&persistence).await;
    let created = persistence
        .enqueue_standalone_delivery(standalone(
            home.company.id,
            DeliveryKey::parse(format!("notification:bounce:{}", Uuid::new_v4())).unwrap(),
        ))
        .await
        .unwrap();
    park(&persistence, created.delivery_id()).await;
    let insert = |company: Option<Uuid>| {
        sqlx::query(
            "INSERT INTO message_delivery_parts (id, company_id, delivery_id, part_index, part_key, payload, content_digest) \
             SELECT gen_random_uuid(), $2, delivery_id, 7, 'email:extra', payload, content_digest \
               FROM message_delivery_parts WHERE delivery_id = $1",
        )
        .bind(created.delivery_id().as_uuid())
        .bind(company)
    };
    let crossed = insert(Some(foreign.company.id))
        .execute(&persistence.pool)
        .await
        .unwrap_err();
    assert_eq!(
        constraint(&crossed),
        Some("message_delivery_parts_delivery_fk"),
        "{crossed}"
    );
    let untenanted = insert(None).execute(&persistence.pool).await.unwrap_err();
    assert_eq!(
        untenanted
            .as_database_error()
            .and_then(|error| error.code())
            .as_deref(),
        Some("23502"),
        "{untenanted}"
    );
    for company in [home.company.id, foreign.company.id] {
        CompanyPersistence::delete(&persistence, company)
            .await
            .unwrap();
    }
}

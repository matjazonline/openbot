    pub(super) async fn tool(
        &self,
        checkpoint: &RunCheckpoint,
        call: SavedToolCall,
    ) -> AppResult<Progress> {
        let invocation = call.invocation_id;
        let prepared = self
            .update(
                checkpoint,
                WorkflowCheckpointUpdate::PrepareInvocation(invocation),
            )
            .await?;
        // Only prepared declarations dispatch; a model-invented name or target never does.
        let Some(tool) = self.prepared.tool(call.tool_id.as_str()) else {
            let refused = json!({"error":{"code":"unknown_tool","message":"Tool is not available to this agent"}});
            return Ok(Progress::Continue(
                self.update(
                    &prepared,
                    WorkflowCheckpointUpdate::InvocationResult {
                        invocation,
                        result: refused,
                    },
                )
                .await?,
            ));
        };
        let id = ModelToolCallId::parse(invocation.0.to_string())?;
        match self
            .tools
            .dispatch(self.fence, tool, id, call.arguments)
            .await?
        {
            AgentToolObservation::Completed(result) => Ok(Progress::Continue(
                self.update(
                    &prepared,
                    WorkflowCheckpointUpdate::InvocationResult { invocation, result },
                )
                .await?,
            )),
            AgentToolObservation::Parked(wait) => Ok(Progress::Suspend(prepared, wait)),
            AgentToolObservation::Uncertain => {
                Err(AppError::Execution(ExecutionFailure::IndeterminateEffect))
            }
        }
    }

async fn saved_call_on(
    db: &mut PgConnection, scope: ActionScope, origin: AgentToolOrigin,
    declared: &AgentModelToolName, call: &ModelToolCallId, arguments: &Value,
) -> AppResult<()> {
    let invocation = Uuid::parse_str(call.as_str()).map_err(|_| missing())?;
    let rows = sqlx::query_as::<_, SavedCall>(
        "SELECT harness.id,harness.checkpoint,harness.workflow_prepared FROM task_harness_runs AS harness \
         WHERE harness.company_id=$1 AND harness.workflow_run_id=$2 AND harness.workflow_execution_id=$3 AND harness.agent_id=$4 \
         AND EXISTS(SELECT 1 FROM jsonb_array_elements(harness.checkpoint->'invocations') AS invocation WHERE invocation->'call'->>'invocation_id'=$5) LIMIT 2"
    ).bind(scope.company.as_uuid()).bind(scope.run.as_uuid()).bind(scope.execution.as_uuid()).bind(origin.agent.as_uuid())
        .bind(invocation.to_string()).fetch_all(db).await?;
    let [row] = rows.as_slice() else { return Err(missing()); };
    let tools: Vec<crate::application::workflow::agent::PreparedTool> = serde_json::from_value(row.workflow_prepared["tools"].clone()).map_err(|_| invalid())?;
    if !tools.iter().any(|tool| tool.origin == origin && &tool.declared == declared) { return Err(missing()); }
    let checkpoint = crate::adapters::persistence::harness_checkpoint::decode(row.id, row.checkpoint.clone())?;
    let saved = checkpoint.invocations.iter().find(|saved| saved.call.invocation_id.0 == invocation).ok_or_else(missing)?;
    if saved.call.tool_id.as_str() != declared.as_str() || &saved.call.arguments != arguments { return Err(missing()); }
    use crate::services::harness::runs::InvocationState;
    match saved.state {
        InvocationState::Completed if saved.result.is_some() => Ok(()),
        InvocationState::Prepared | InvocationState::Ready | InvocationState::Indeterminate
            if checkpoint.pending_call().is_some_and(|pending| pending.invocation_id.0 == invocation) => Ok(()),
        _ => Err(missing()),
    }
}

async fn assistant_channel_commands_race_to_one_assistant() {
    let f = command_fixture().await;
    let before = written(&f).await;
    let barrier = Barrier::new(2);
    let (a, b) = tokio::join!(
        async {
            barrier.wait().await;
            service(&f)
                .create_assistant_channel(company(&f), owner(&f), command("same", "same"))
                .await
        },
        async {
            barrier.wait().await;
            service(&f)
                .create_assistant_channel(company(&f), owner(&f), command("same", "same"))
                .await
        }
    );
    assert_eq!(a.unwrap().unwrap(), b.unwrap().unwrap());
    assert_eq!(grew(&before, &written(&f).await)["bindings"], 1);

    let (c, d) = tokio::join!(
        async {
            barrier.wait().await;
            service(&f)
                .create_assistant_channel(company(&f), owner(&f), command("one", "contested"))
                .await
        },
        async {
            barrier.wait().await;
            service(&f)
                .create_assistant_channel(company(&f), owner(&f), command("two", "contested"))
                .await
        }
    );
    let outcomes = [c.unwrap(), d.unwrap()];
    assert_eq!(outcomes.iter().filter(|outcome| outcome.is_ok()).count(), 1);
    assert!(
        outcomes
            .iter()
            .any(|outcome| matches!(outcome, Err(ProvisioningRefusal::Conflict(_))))
    );
    assert_eq!(grew(&before, &written(&f).await)["bindings"], 2);
    let records: i64 = sqlx::query_scalar(
        "SELECT count(*) FROM workflow_provisioning_commands WHERE company_id = $1",
    )
    .bind(company(&f).as_uuid())
    .fetch_one(f.persistence().pool())
    .await
    .unwrap();
    assert_eq!(records, 2, "a refused command records nothing");
}

pub async fn ui_create_assistant_channel(
    State(channels): State<Arc<dyn AssistantChannels>>,
    State(agents): State<Arc<AgentUseCases>>,
    user: AuthenticatedUser,
    Path(agent_id): Path<Uuid>,
    Query(query): Query<CompanyQuery>,
    Form(form): Form<UiAssistantChannelForm>,
) -> AppResult<Html<String>> {
    let agent = agents
        .get_company_agent(user.id, query.company_id, agent_id)
        .await?
        .ok_or_else(|| AppError::NotFound("Agent not found".into()))?;
    let created = create(
        channels.as_ref(),
        &user,
        query.company_id,
        AssistantChannelRequest {
            agent_slug: agent.slug.clone(),
            channel_slug: form.channel_slug,
            channel_name: form.channel_name,
            reply_mode: form.reply_mode,
            command_key: crate::adapters::http::routes::schedule::run_now_key(form.command_key),
        },
    )
    .await;
    let card = match created {
        Ok(created) => pages::assistant_channel_card(
            query.company_id,
            &agent,
            Some(pages::AssistantChannelOutcome::Created {
                channel: created["channel"]["slug"].as_str().unwrap_or_default(),
                address: created["channel"]["address"].as_str().unwrap_or_default(),
            }),
        ),
        Err(AppError::BadRequest(message) | AppError::Conflict(message)) => {
            pages::assistant_channel_card(
                query.company_id,
                &agent,
                Some(pages::AssistantChannelOutcome::Refused(&message)),
            )
        }
        Err(error) => return Err(error),
    };
    Ok(Html(card))
}

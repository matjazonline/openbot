        .await?

    if !current.enabled {
        return Err(unavailable());
    }
    if current.auth == McpAuth::Bearer {
        credential_ready(tx, company, connection, CredentialOwner::Mcp).await?;

    sqlx::query_scalar::<_, String>(

    .bind(connection)

    .fetch_optional(&mut *tx)
    .await?

    let reviewed: ReviewedMcpOperation = sqlx::query_as(
        "SELECT snapshot,approval_required,version FROM workflow_mcp_operations WHERE company_id=$1 \
         AND connection_id=$2 AND name=$3 AND revoked_at IS NULL FOR SHARE"
    ).bind(company.as_uuid()).bind(connection).bind(operation.as_str())
        .fetch_optional(&mut *tx).await?.ok_or_else(unavailable)?;
    if reviewed.version != reviewed_version {

    }
    let tool: ToolSnapshot = serde_json::from_value(reviewed.snapshot).map_err(|_| invalid())?;
    if tool.contract.name.as_str() != remote_name {
        return Err(unavailable());
    }
    McpOperation {
        tool: tool.clone(),
        approval_required: reviewed.approval_required,
    }
    .validate(
        company,
        current
            .discovered_tools
            .iter()
            .find(|tool| tool.name.as_str() == remote_name),
        current
            .tool_grants
            .iter()
            .any(|name| name.as_str() == remote_name),
    )?;
    Ok(RegisteredTool {
        tool,
        approval_required: reviewed.approval_required,
    })

pub(crate) fn validate_catalog(
    discovered_tools: &[McpDiscoveredTool],
    tool_grants: &[McpToolName],
) -> AppResult<()> {
    if discovered_tools.len() > MAX_MCP_DISCOVERED_TOOLS
        || tool_grants.len() > MAX_EFFECTIVE_MCP_TOOLS
    {
        return Err(invalid("MCP tool count exceeds the server limit"));
    }
    let encoded =
        serde_json::to_vec(discovered_tools).map_err(|_| invalid("Invalid MCP discovery"))?;
    if encoded.len() > MAX_MCP_DISCOVERY_BYTES {
        return Err(invalid("MCP discovery exceeds 1 MiB"));
    }
    let mut names = std::collections::HashSet::new();
    for tool in discovered_tools {
        if !names.insert(&tool.name) {
            return Err(invalid("Duplicate discovered MCP tool name"));
        }
        for schema in std::iter::once(&tool.input_schema).chain(&tool.output_schema) {
            crate::services::tool_schema::compile(schema)?;
            if !schema.is_object()
                || serde_json::to_vec(schema)
                    .map_err(|_| invalid("Invalid MCP schema"))?
                    .len()
                    > MAX_MCP_SCHEMA_BYTES
            {
                return Err(invalid(
                    "MCP tool schema must be an object of at most 64 KiB",
                ));
            }
        }
    }
    let mut grants = std::collections::HashSet::new();
    for grant in tool_grants {
        if !names.contains(grant) || !grants.insert(grant) {
            return Err(invalid(
                "MCP grants must uniquely name reviewed discovered tools",
            ));
        }
    }
    Ok(())
}

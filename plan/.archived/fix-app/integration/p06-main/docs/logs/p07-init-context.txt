fn security_session_metrics_export_under_startup_fly_and_example_filters() {
    for filter in [
        DEFAULT_LOG_FILTER,
        configured_filter(include_str!("../../fly.toml")),
        configured_filter(include_str!("../../.env.example")),
    ] {
        let output = Arc::new(Mutex::new(Vec::new()));
        let sink = output.clone();
        let subscriber = tracing_subscriber::registry()
            .with(EnvFilter::new(filter))
            .with(
                fmt::layer()
                    .without_time()
                    .with_ansi(false)
                    .with_target(false)
                    .with_writer(move || Capture(sink.clone())),
            );
        tracing::subscriber::with_default(subscriber, || {
            let monitor = TracingMonitor::new();
            monitor.increment_counter(
                "session_validation_failures",
                1,
                &[("reason", "store_unavailable")],
            );
            monitor.record_gauge("session_cleanup_backlog", 1001.0, &[]);
            tracing::info!(target: "open_bots::session_probe", "application-marker");
            tracing::info!(target: "monitoring::smtp", "excluded-smtp-marker");
            tracing::info!(target: "monitoring::ai", "excluded-ai-marker");
        });
        let captured = String::from_utf8(output.lock().unwrap().clone()).unwrap();
        assert!(
            captured.contains("application-marker"),
            "{filter}: {captured}"
        );
        assert!(
            captured.contains("session_validation_failures"),
            "{filter}: {captured}"
        );
        assert!(
            captured.contains("store_unavailable"),
            "{filter}: {captured}"
        );
        assert!(
            captured.contains("session_cleanup_backlog"),
            "{filter}: {captured}"
        );
        assert!(!captured.contains("excluded-smtp-marker"));
        assert!(!captured.contains("excluded-ai-marker"));
    }
}

    async fn a_fresh_database_applies_only_the_init_schema() {
        let migrator = sqlx::migrate!("./migrations");
        let expected: Vec<_> = migrator
            .iter()
            .map(|migration| {
                (
                    migration.version,
                    migration.description.to_string(),
                    migration.checksum.to_vec(),
                )
            })
            .collect();
        assert_eq!(
            expected
                .iter()
                .map(|(version, description, _)| (*version, description.as_str()))
                .collect::<Vec<_>>(),
            vec![(20260817000000, "init schema")]
        );
        let Some(database) = own_database().await else {
            return;
        };
        let ledger = || async {
            sqlx::query_as::<_, (i64, String, Vec<u8>, bool)>(
                "SELECT version, description, checksum, success FROM _sqlx_migrations ORDER BY version",
            )
            .fetch_all(&database.pool)
            .await
            .unwrap()
        };
        let recorded = ledger().await;
        assert_eq!(
            recorded
                .iter()
                .map(|(version, description, checksum, success)| {
                    assert!(*success);
                    (*version, description.clone(), checksum.clone())
                })
                .collect::<Vec<_>>(),
            expected
        );
        migrator.run(&database.pool).await.unwrap();
        assert_eq!(ledger().await, recorded, "a rerun applies nothing");
        let populated: Vec<String> = sqlx::query_scalar(
            "SELECT relname::text FROM pg_class WHERE relnamespace='public'::regnamespace AND relkind IN ('r','p') \
               AND relname<>'_sqlx_migrations' AND (xpath('/row/n/text()', query_to_xml(format('SELECT count(*) AS n FROM public.%I', relname), false, true, '')))[1]::text<>'0'",
        )
        .fetch_all(&database.pool)
        .await
        .unwrap();
        assert!(
            populated.is_empty(),
            "the init schema seeds no rows: {populated:?}"
        );
    }

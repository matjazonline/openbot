    async fn usage_report(&self, query: UsageQuery) -> AppResult<UsageReport> {
        query.validate()?;
        let mut tx = self.pool.begin().await?;
        authorize(&mut tx, query.scope).await?;
        validate_filters(&mut tx, &query).await?;
        let mut calls = load_calls(&mut tx, &query).await?;
        let truncated = calls.len() > MAX_CALLS;
        calls.truncate(MAX_CALLS);
        let scanned_calls = calls.len();
        let mut prices: Vec<PriceRow> = sqlx::query_as(
            "SELECT provider,model,input_micro_usd_per_million AS input,
             output_micro_usd_per_million AS output,revision FROM model_prices
             WHERE company_id=$1 ORDER BY provider,model LIMIT $2",
        )
        .bind(query.scope.company)
        .bind((MAX_PRICES + 1) as i64)
        .fetch_all(&mut *tx)
        .await?;
        let prices_truncated = prices.len() > MAX_PRICES;
        prices.truncate(MAX_PRICES);
        let (groups, groups_truncated) = aggregate(
            calls
                .into_iter()
                .map(TryInto::try_into)
                .collect::<AppResult<Vec<_>>>()?,
            &query,
        )?;
        tx.commit().await?;
        Ok(UsageReport {
            start: query.start,
            end: query.end,
            grouping: query.grouping,
            groups,
            scanned_calls,
            truncated: truncated || groups_truncated,
            prices: prices
                .into_iter()
                .map(TryInto::try_into)
                .collect::<AppResult<_>>()?,
            prices_truncated,
        })
    }

    // The parent's own parameter is valid for the parent and too long for the child. The
    // step's `params` contract embeds the child's parameter schema, so activation refuses the
    // frozen inputs: like any step's invalid input, the run fails unrouted and no child exists.
    // (A refusal after activation takes the error route: the child-limits case.)
    let (f, root) = AdmissionFixture::with_bound_child(
        leaf(),
        parent,
        json!({"reviewer":"far-too-long"}),
        &["service", "archive"],
    )
    .await;

    assert!(p.park_child(root).await.unwrap().is_none());
    let observed = || async {
        sqlx::query_as::<_, (String, i64, i64, i64)>(
            "SELECT (SELECT state FROM workflow_runs WHERE id=$1), \
             (SELECT count(*) FROM workflow_runs WHERE parent_run_id=$1), \
             (SELECT count(*) FROM workflow_run_events WHERE run_id=$1 AND event_kind='workflow.invalid_input'), \
             (SELECT count(*) FROM workflow_executions WHERE run_id=$1 AND step_id='recover')",

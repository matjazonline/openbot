    pub async fn scan_mime(&self, raw_mime: &[u8]) -> SpamScanResult {
        let result = SpamScanResult::new();

        if !self.config.enable_spam_scanner {
            return result;
        }

        match self.config.spam_scanner_type.to_lowercase().as_str() {
            "spamassassin" | "spamd" => match self.scan_spamassassin(raw_mime).await {
                Ok(res) => return res,
                Err(err) => {
                    warn!("SpamAssassin scan failed: {err}");
                }
            },
            _ => {
                // Default to Rspamd HTTP
                match self.scan_rspamd(raw_mime).await {
                    Ok(res) => return res,
                    Err(err) => {
                        warn!("Rspamd scan failed: {err}");
                    }
                }
            }
        }

        result
    }

    async fn apply_spam_score(&self, session: &SmtpSession, raw_payload: &mut RawInboundPayload) {
        let scanner = crate::services::spam_scanner::SpamScannerService::new(self.config.clone());
        let scan_res = scanner
            .scan(
                &session.data_buffer,
                raw_payload.subject.as_deref(),
                Some(&raw_payload.from),
                raw_payload.text.as_deref(),
                raw_payload.html.as_deref(),
            )
            .await;

        if scan_res.score > 0.0 {
            raw_payload.spam_score = Some(scan_res.score);
            if let Some(ref m) = self.monitoring {
                m.record_histogram("smtp_spam_score", scan_res.score, &[]);
            }
        }
    }

fn security_session_metrics_export_under_startup_fly_and_example_filters() {
    for filter in [
        DEFAULT_LOG_FILTER,
        configured_filter(include_str!("../../fly.toml")),
        configured_filter(include_str!("../../.env.example")),
    ] {
        let output = Arc::new(Mutex::new(Vec::new()));
        let sink = output.clone();
        let subscriber = tracing_subscriber::registry()
            .with(EnvFilter::new(filter))
            .with(
                fmt::layer()
                    .without_time()
                    .with_ansi(false)
                    .with_target(false)
                    .with_writer(move || Capture(sink.clone())),
            );
        tracing::subscriber::with_default(subscriber, || {
            let monitor = TracingMonitor::new();
            monitor.increment_counter(
                "session_validation_failures",
                1,
                &[("reason", "store_unavailable")],
            );
            monitor.record_gauge("session_cleanup_backlog", 1001.0, &[]);
            tracing::info!(target: "open_bots::session_probe", "application-marker");
            tracing::info!(target: "monitoring::smtp", "excluded-smtp-marker");
            tracing::info!(target: "monitoring::ai", "excluded-ai-marker");
        });
        let captured = String::from_utf8(output.lock().unwrap().clone()).unwrap();
        assert!(
            captured.contains("application-marker"),
            "{filter}: {captured}"
        );
        assert!(
            captured.contains("session_validation_failures"),
            "{filter}: {captured}"
        );
        assert!(
            captured.contains("store_unavailable"),
            "{filter}: {captured}"
        );
        assert!(
            captured.contains("session_cleanup_backlog"),
            "{filter}: {captured}"
        );
        assert!(!captured.contains("excluded-smtp-marker"));
        assert!(!captured.contains("excluded-ai-marker"));
    }
}

    pub(crate) async fn binding_status(
        &self,
        request: BindingStatusRequest<'_>,
    ) -> IntegrationResult<Option<AccountUseStatus>> {
        let mut tx = self
            .persistence
            .pool()
            .begin()
            .await
            .map_err(storage_error)?;
        sqlx::query("SET LOCAL lock_timeout='5s'")
            .execute(&mut *tx)
            .await
            .map_err(storage_error)?;
        let grant = match super::grants::grant_on(&mut tx, request.reference.grant).await {
            Ok(grant) => grant,
            Err(IntegrationError::NotFound) => return Ok(None),
            Err(error) => return Err(error),
        };
        if grant.account != request.reference.account
            || &grant.client != request.client
            || &grant.host_scope != request.scope
            || &grant.binding != request.binding
        {
            return Err(IntegrationError::Denied);
        }
        let account =
            match super::grants::read_account_on(&mut tx, &grant.owner, grant.account).await {
                Ok(account) => account,
                Err(IntegrationError::NotFound) => return Ok(None),
                Err(error) => return Err(error),
            };
        let grant = super::grants::grant_on(&mut tx, grant.id).await?;
        if grant.account != request.reference.account
            || &grant.client != request.client
            || &grant.host_scope != request.scope
            || &grant.binding != request.binding
        {
            return Err(IntegrationError::Denied);
        }
        let policy_current = match self.registry.policy(&grant.app) {
            Ok(policy) => policy == grant.app_policy,
            Err(IntegrationError::NotFound) => false,
            Err(error) => return Err(error),
        };
        let result = AccountUseStatus {
            account: account.id,
            app: grant.app,
            label: account.metadata.label().to_owned(),
            status: account.status,
            authorization_current: policy_current
                && request.reference.authorization == account.authorization_revision
                && grant.account_authorization == account.authorization_revision
                && request.reference.grant_revision == grant.revision,
            grant_enabled: grant.enabled,
        };
        tx.rollback().await.map_err(storage_error)?;
        Ok(Some(result))
    }

    async fn status_once(&self, read: &ClientRead, operation: Uuid) -> TxResult<GenerationStatus> {
        let mut tx = self.begin().await?;
        sqlx::query!("SET TRANSACTION ISOLATION LEVEL REPEATABLE READ, READ ONLY")
            .execute(&mut *tx)
            .await?;
        let row = sqlx::query!(
            "SELECT o.state, o.outcome_code FROM generation_operations AS o \
               JOIN namespaces AS n ON (n.tenant_id, n.id) = (o.tenant_id, o.namespace_id) \
              WHERE o.tenant_id = $1 AND o.id = $2 AND o.namespace_id = ANY($3) \
                AND n.state = 'active' AND o.input_level <= $4::smallint",
            read.tenant_id,
            operation,
            &read.namespace_ids,
            read.read_ceiling.as_i16(),
        )
        .fetch_optional(&mut *tx)
        .await?
        .ok_or_else(not_found)?;
        let observations = sqlx::query_scalar!(
            "SELECT id FROM observations WHERE tenant_id = $1 AND generation_operation_id = $2 \
                AND access_level <= $3::smallint AND (expires_at IS NULL OR expires_at > now()) \
              ORDER BY output_ordinal",
            read.tenant_id,
            operation,
            read.read_ceiling.as_i16(),
        )
        .fetch_all(&mut *tx)
        .await?;
        tx.commit().await?;
        // A pending cancel reason waits in `outcome_code`; it is an outcome only once final.
        let state = status_of(&row.state)?;
        let outcome = match state {
            GenerationState::Accepted | GenerationState::Claimed | GenerationState::Dispatched => {
                None
            }
            _ => row.outcome_code,
        };
        Ok(GenerationStatus {
            operation_id: operation,
            state,
            outcome,
            observations,
            replayed: false,
        })
    }

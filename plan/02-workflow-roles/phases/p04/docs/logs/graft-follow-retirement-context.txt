async fn department_desk_competing_removal_and_demotion_of_one_principal_both_commit() {
    let Some(database) = own_database().await else {
        return;
    };
    let f = Arc::new(Departments::on(database.pool.clone()).await);
    let unit = f.create("Pair", None).await;
    add(&f, unit.id, f.member_principal).await;
    lead(&f, unit.id, f.member_principal).await;
    let room = desk(&f, "pair-room", owned(unit.id, true)).await;
    assert_eq!(derived(&room), both(&[f.member_principal]));
    sqlx::query(&format!(
        r#"CREATE FUNCTION pause_department_removal() RETURNS trigger LANGUAGE plpgsql AS $$
           BEGIN IF OLD.principal_id = '{}'::uuid THEN PERFORM pg_advisory_xact_lock(61018); END IF;
           RETURN OLD; END $$"#,
        f.member_principal
    ))
    .execute(f.pool())
    .await
    .unwrap();
    sqlx::query(
        "CREATE TRIGGER zz_pause_department_removal AFTER DELETE ON department_members \
         FOR EACH ROW EXECUTE FUNCTION pause_department_removal()",
    )
    .execute(f.pool())
    .await
    .unwrap();
    let mut gate = f.pool().begin().await.unwrap();
    sqlx::query("SELECT pg_advisory_xact_lock(61018)")
        .execute(&mut *gate)
        .await
        .unwrap();
    let removal = tokio::spawn({
        let f = f.clone();
        async move {
            f.use_cases
                .remove_member(f.owner, f.company, unit.id, f.member_principal)
                .await
        }
    });
    wait_until_backends_are_blocked(f.pool(), 1).await;
    let demotion = tokio::spawn({
        let pool = f.pool().clone();
        let principal = f.member_principal;
        async move {
            sqlx::query("UPDATE principals SET kind = 'external', user_id = NULL WHERE id = $1")
                .bind(principal.as_uuid())
                .execute(&pool)
                .await
        }
    });
    wait_until_backends_are_blocked(f.pool(), 2).await;
    gate.commit().await.unwrap();
    removal.await.unwrap().expect("the removal commits");
    demotion
        .await
        .unwrap()
        .expect("the demotion commits, no deadlock");
    let after = reload(&f, room.id).await;
    assert_eq!(
        derived(&after),
        Vec::new(),

async fn memory_boundary_separate_database_roles_cannot_cross() {
    let db = test_db().await;
    let outsider = format!("{}_other", db.name);
    let password = format!("{:x}", uuid::Uuid::new_v4().as_u128());
    admin_sql(&format!(
        "CREATE ROLE \"{outsider}\" LOGIN PASSWORD '{password}'"
    ))
    .await;
    let as_outsider = db
        .owner_options
        .clone()
        .username(&outsider)
        .password(&password);
    // PUBLIC has no CONNECT on a service database: another application's role is refused.
    let refused = PgConnection::connect_with(&as_outsider)
        .await
        .unwrap_err()
        .to_string();
    assert!(refused.contains("permission denied"), "{refused}");
    // Even with CONNECT, it cannot read the service's tables.
    admin_sql(&format!(
        "GRANT CONNECT ON DATABASE \"{}\" TO \"{outsider}\"",
        db.name
    ))
    .await;
    let mut conn = PgConnection::connect_with(&as_outsider).await.unwrap();
    let denied = sqlx::query("SELECT count(*) FROM tenants")
        .execute(&mut conn)
        .await
        .unwrap_err()
        .to_string();
    assert!(denied.contains("permission denied"), "{denied}");
    conn.close().await.unwrap();
    // Positive control: the owner role reads its own tables.
    assert!(
        sqlx::query("SELECT count(*) FROM tenants")
            .execute(db.pool())
            .await
            .is_ok()
    );
    admin_sql(&format!(
        "REVOKE CONNECT ON DATABASE \"{}\" FROM \"{outsider}\"",
        db.name
    ))
    .await;
    db.pool().close().await;
    drop_database(&db.name);
    admin_sql(&format!("DROP ROLE IF EXISTS \"{outsider}\"")).await;
}

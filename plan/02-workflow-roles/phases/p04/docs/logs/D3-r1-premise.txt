48: /// How a transaction takes the gate, which is also why.
49: #[derive(Debug, Clone, Copy, PartialEq, Eq)]
50: pub(crate) enum ChannelGateAccess {
51:     /// Admitting a message through the channel's current settings. Any number may hold it at once.
52:     Admit,
53:     /// Changing those settings. Waits for every admission in flight, and holds new ones back
54:     /// until it commits.
55:     ChangeSettings,
56: }
57: 
58: /// Take the gate for every key, in `(company_id, channel_id)` order.
59: ///
60: /// A multi-channel writer must pass all of its keys in one call: taking them one at a time in the
61: /// order it happens to meet them is how two such writers deadlock each other.
62: pub(crate) async fn acquire_channel_gates_on(
63:     conn: &mut PgConnection,
64:     keys: impl IntoIterator<Item = ChannelGateKey>,
65:     access: ChannelGateAccess,
66: ) -> AppResult<()> {
67:     let mut keys: Vec<ChannelGateKey> = keys.into_iter().collect();
68:     keys.sort_unstable();
69:     keys.dedup();
70:     let function = match access {
71:         ChannelGateAccess::Admit => "pg_advisory_xact_lock_shared",
72:         ChannelGateAccess::ChangeSettings => "pg_advisory_xact_lock",
73:     };
74:     let statement = format!(
75:         "SELECT {function}(hashtextextended('channel-gate:' || $1::text || ':' || $2::text, 0))"
76:     );
77:     for key in keys {
78:         sqlx::query(&statement)
79:             .bind(key.company_id)
80:             .bind(key.channel_id)
81:             .execute(&mut *conn)
82:             .await
83:             .map_err(AppError::from)?;
84:     }
85:     Ok(())
86: }

93: pub(super) async fn commit_on(
94:     tx: &mut Transaction<'_, Postgres>,
95:     request: &InboundCommitRequest,
96: ) -> AppResult<InboundCommitOutcome> {
97:     if request.associations.is_empty() {
98:         return Err(AppError::BadRequest(
99:             "An inbound commit must name at least one thread association".into(),
100:         ));
101:     }
102:     hold_channel_settings(tx, request).await?;
103:     let binding_ids = verify_association_bindings(tx, request).await?;
104:     lock_provider_keys(tx, request, &binding_ids).await?;
105: 
106:     // Asked first, and under every relevant lock. A redelivery must open no thread: the old path
107:     // created the thread before it knew whether the message was new, so a provider retry left an
108:     // empty conversation behind and then failed associating the message it already held.
109:     let existing = existing_message_mappings(tx, request, &binding_ids).await?;
110:     if !existing.is_empty() {
111:         return recognise_redelivery(tx, request, existing).await;
112:     }
113: 
114:     let threads = resolve_threads(tx, request).await?;
115:     bind_conversations(tx, request, &threads).await?;
116: 
117:     let write = message_write(request, threads[0].thread_id)?;
118:     let stored = store_canonical(tx, request, &write).await?;
119: 
120:     associate_threads(tx, request, &threads, stored.id).await?;
121:     map_provider_message(tx, request, &threads, &stored).await?;
122:     // After the associations, whose rows the intents' foreign key names, and before the fan-out:
123:     // keeping the statement order of the whole commit fixed is what stops two concurrent inbound
124:     // commits on one thread taking their locks in opposite orders.
125:     let intent_ids = create_intents(tx, request, &threads, stored.id).await?;
126:     let delivery_ids = create_deliveries(tx, request).await?;
127:     complete_claimed_event(tx, request).await?;
128: 
129:     Ok(InboundCommitOutcome {
130:         disposition: CommitDisposition::Created,
131:         message_id: stored.id,
132:         thread_ids: threads.iter().map(|thread| thread.thread_id).collect(),
133:         intent_ids,
134:         delivery_ids,
135:     })
136: }

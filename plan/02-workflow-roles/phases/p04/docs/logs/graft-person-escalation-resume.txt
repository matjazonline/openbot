pub(crate) async fn case_scope_on(
    db: &mut PgConnection,
    company: CompanyId,
    case: EscalationId,
) -> AppResult<Option<CaseScope>> {
    let row=sqlx::query_as::<_,ScopeRow>("SELECT escalation.source_kind,escalation.run_id,run.correlation_id, \
        COALESCE(run.channel_id,refusal.channel_id) AS channel_id,COALESCE(run.thread_id,refusal.thread_id) AS thread_id,refusal.source_key \
        FROM workflow_escalation_cases AS escalation \
        LEFT JOIN workflow_runs AS run ON run.company_id=escalation.company_id AND run.id=escalation.run_id \
        LEFT JOIN workflow_admission_refusals AS refusal ON refusal.company_id=escalation.company_id AND refusal.id=escalation.admission_refusal_id \
        WHERE escalation.company_id=$1 AND escalation.id=$2")
        .bind(company.as_uuid()).bind(case.as_uuid()).fetch_optional(db).await?;
    let Some(row) = row else { return Ok(None) };
    let payload = match row.source_kind.as_str() {
        "run" | "empty_reviewers" => {
            PayloadSource::Run(RunId::new(row.run_id.ok_or_else(invalid)?))
        }
        "admission" => match admission_binding::SourceKey::payload_source(
            row.source_key.as_deref().ok_or_else(invalid)?,
        )? {
            admission_binding::AdmissionPayloadSource::Message(message) => {
                PayloadSource::Message(message)
            }
            admission_binding::AdmissionPayloadSource::Public => PayloadSource::Public,
        },
        _ => return Err(invalid()),
    };
    Ok(Some(CaseScope {
        company,
        payload,
        run_id: row.run_id,
        channel_id: row.channel_id,
        thread_id: row.thread_id,
        correlation_id: row.correlation_id,
    }))
}

async fn recipient_for_source(
    connection: &mut PgConnection,
    event: &NotificationEvent,
    source: SourceSnapshot,
    principal_id: Uuid,
) -> AppResult<Option<NotificationRecipient>> {
    #[derive(sqlx::FromRow)]
    struct RecipientRow {
        user_id: Uuid,
        email: String,
        company_label: String,
        channel_label: String,
        delivery_failure_email_enabled: bool,
        decision_email_enabled: bool,
    }
    let row = sqlx::query_as::<_, RecipientRow>(
        r#"SELECT principal.user_id, account.email,
                  company.name AS company_label, COALESCE(channel.name,'Company workflow') AS channel_label,
                  COALESCE(preference.delivery_failure_email_enabled, TRUE)
                      AS delivery_failure_email_enabled,
                  COALESCE(preference.decision_email_enabled, TRUE)
                      AS decision_email_enabled
             FROM principals AS principal
             JOIN users AS account ON account.id = principal.user_id
             JOIN company_members AS member
               ON member.company_id = principal.company_id
              AND member.user_id = principal.user_id
             JOIN companies AS company ON company.id = principal.company_id
             LEFT JOIN channels AS channel
               ON channel.company_id = principal.company_id AND channel.id = $3
             LEFT JOIN user_notification_preferences AS preference
               ON preference.user_id = account.id
            WHERE principal.company_id = $1 AND principal.id = $2
              AND principal.kind = 'person'
              AND notification_principal_can_view($1, $3, $2)
            "#,
    )
    .bind(event.company_id)
    .bind(principal_id)
    .bind(source.channel_id)
    .fetch_optional(&mut *connection)
    .await
    .map_err(AppError::from)?;
    Ok(row.map(|row| {
        let preferences = NotificationPreferences {
            delivery_failure_email_enabled: row.delivery_failure_email_enabled,
            decision_email_enabled: row.decision_email_enabled,
        };
        NotificationRecipient {
            principal_id: PrincipalId::new(principal_id),
            user_id: row.user_id,
            email: EmailAddress::from(row.email),
            company_label: row.company_label,
            channel_label: row.channel_label,
            channel_id: source.channel_id,
            thread_id: source.thread_id,
            run_id: source.run_id,
            correlation_id: source
                .correlation_id
                .map(Into::into)
                .unwrap_or_else(|| event.id.as_uuid().into()),
            email_enabled: event.action_kind.email_preference(preferences),
            target: source.target,
        }
    }))
}

pub(super) async fn source_on(
    db: &mut PgConnection,
    event: &NotificationEvent,
) -> AppResult<Option<SourceSnapshot>> {
    let source:Option<ReviewerSource>=sqlx::query_as(
        "SELECT notice.decision_id,run.channel_id,run.thread_id,run.id AS run_id,run.correlation_id,notice.principal_id, \
         workflow_follow_is_open(notice.company_id,notice.decision_id) AS open, \
         EXISTS (SELECT 1 FROM workflow_decision_reviewers AS current \
             WHERE current.company_id=notice.company_id AND current.decision_id=notice.decision_id \
               AND current.principal_id=notice.principal_id AND current.notice_event_id=notice.event_id \
               AND current.notice_assignment_revision=notice.assignment_revision) AND workflow_follow_reviewer_eligible(notice.company_id,run.id,notice.principal_id) AS current_reviewer \
         FROM workflow_reviewer_notices AS notice \
         JOIN workflow_decisions AS decision ON decision.company_id=notice.company_id AND decision.id=notice.decision_id \
         JOIN workflow_runs AS run ON run.company_id=decision.company_id AND run.id=decision.run_id \
         WHERE notice.company_id=$1 AND notice.id=$2 AND notice.event_id=$3 AND notice.assignment_revision=$4")
        .bind(event.company_id).bind(event.source_id).bind(event.id.as_uuid())
        .bind(i64::try_from(event.source_generation).map_err(|_|AppError::BadRequest("Invalid notice generation".into()))?)
        .fetch_optional(db).await?;
    Ok(source.map(|source| SourceSnapshot {
        channel_id: source.channel_id,
        thread_id: source.thread_id,
        run_id: Some(source.run_id),
        correlation_id: Some(source.correlation_id),
        responsible_principal_id: Some(source.principal_id),
        target: NotificationTarget::WorkflowDecision {
            decision_id: source.decision_id,
        },
        lifecycle: if !source.open {
            SourceLifecycle::Resolved
        } else if !source.current_reviewer {
            SourceLifecycle::Withdrawn
        } else {
            SourceLifecycle::Active
        },
    }))
}

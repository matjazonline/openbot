    async fn commit_notification_projection(
        &self,
        commit: NotificationProjectionCommit<'_>,
    ) -> AppResult<(
        NotificationProjectionCommitOutcome,
        NotificationProjectionResult,
    )> {
        let mut tx = self.pool.begin().await.map_err(AppError::from)?;
        let Some(hint) = read_claimed_event(&mut tx, commit.fence).await? else {
            return Ok((
                NotificationProjectionCommitOutcome::LeaseLost,
                NotificationProjectionResult::default(),
            ));
        };
        let prepared =
            projection_lock::lock_on(&mut tx, &hint, projection_lock::ProjectionPurpose::Project)
                .await?;
        let Some(event) = locked_event(&mut tx, commit.fence).await? else {
            return Ok((
                NotificationProjectionCommitOutcome::LeaseLost,
                NotificationProjectionResult::default(),
            ));
        };
        let current = match prepared {
            projection_lock::LockedProjection::Current(current)
                if current.event == event && *current == *commit.expected =>
            {
                Some(current)
            }
            _ => None,
        };
        let Some(current) = current else {
            release_stale_projection(&mut tx, commit.fence).await?;
            tx.commit().await?;
            return Ok((
                NotificationProjectionCommitOutcome::Stale,
                NotificationProjectionResult::default(),
            ));
        };

        let result = apply_projection_on(&mut tx, &current, commit.email).await?;
        let completed = sqlx::query(
            r#"UPDATE notification_events
                  SET status = 'projected', projected_at = CURRENT_TIMESTAMP,
                      execution_id = NULL, owner_worker_id = NULL,
                      locked_at = NULL, lock_expires_at = NULL
                WHERE id = $1 AND status = 'processing'
                  AND execution_id = $2 AND owner_worker_id = $3 AND lock_expires_at > clock_timestamp()"#,
        )
        .bind(commit.fence.row.as_uuid())
        .bind(commit.fence.execution.as_uuid())
        .bind(commit.fence.owner.as_uuid())
        .execute(&mut *tx)
        .await
        .map_err(AppError::from)?;
        if completed.rows_affected() != 1 {
            return Ok((
                NotificationProjectionCommitOutcome::LeaseLost,
                NotificationProjectionResult::default(),
            ));
        }
        reviewer_notice::record_receipt_on(&mut tx, &current).await?;
        tx.commit().await.map_err(AppError::from)?;
        Ok((NotificationProjectionCommitOutcome::Applied, result))
    }

async fn workflow_control_cancel_resume_competitors_preserve_event_and_notification() {
    for _ in 0..3 {
        let (f, scope) = fixture_source(waiting_source()).await;
        f.persistence().park_wait(scope).await.unwrap().unwrap();
        recorded(f.persistence(), wait_signal(scope)).await.unwrap();
        let before = control_state(&f).await;
        let cmd = command(&f, scope, "cancel").await;
        let barrier = Barrier::new(2);
        let cancel = async {
            barrier.wait().await;
            f.persistence().cancel(cmd).await.unwrap()
        };
        let resume = async {
            barrier.wait().await;
            f.persistence().resume_wait(scope).await.unwrap()
        };
        let (cancel, resume) = tokio::join!(cancel, resume);
        let state = control_state(&f).await;
        if matches!(cancel, CancelResult::Applied { .. }) {
            assert!(matches!(resume, WaitProgress::Refused));
            assert_eq!(state["workflow_executions"].as_array().unwrap().len(), 1);
            assert_eq!(state["workflow_waits"][0]["state"], "cancelled");
        } else {
            assert!(matches!(cancel, CancelResult::RevisionConflict { .. }));
            assert!(matches!(resume, WaitProgress::Completed(_)));
            assert_eq!(state["workflow_executions"].as_array().unwrap().len(), 2);
            assert_eq!(state["workflow_waits"][0]["state"], "completed");
        }
        let mut expected_events = before["workflow_wait_events"].clone();
        if state["workflow_waits"][0]["state"] == "completed" {
            expected_events[0]["consumed_execution_id"] = json!(scope.execution.as_uuid());
        }
        assert_eq!(state["workflow_wait_events"], expected_events);
        assert_eq!(
            state["workflow_waits"][0]["notification_intent"],
            before["workflow_waits"][0]["notification_intent"]
        );
        assert_eq!(state["workflow_waits"].as_array().unwrap().len(), 1);
    }
}

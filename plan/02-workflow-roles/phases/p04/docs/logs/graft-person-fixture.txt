async fn migrated_own_database() -> Option<OwnDatabase> {
    let url = test_database_url()?;
    // Generated, so it needs no quoting beyond the identifier quotes: `CREATE DATABASE` takes no
    // parameters.
    let name = format!("open_bots_own_{}", Uuid::new_v4().simple());
    let admin_url = with_database_name(&url, "postgres");
    admin_statement(&admin_url, &format!(r#"CREATE DATABASE "{name}""#)).await;

    // Every isolated test holds a pool of its own and `cargo test` runs them in parallel, so these
    // are kept small: the default of ten would put a busy machine's worth of tests within reach of
    // the server's `max_connections`. A test that needs more fan-out than this opens its own pool.
    let pool = pool_options()
        .max_connections(5)
        .connect(&with_database_name(&url, &name))
        .await
        .expect("a database this test just created accepts connections");
    let database = OwnDatabase {
        pool,
        name,
        admin_url,
    };
    Box::pin(migrate_test_database(&database.pool))
        .await
        .expect("a fresh database accepts the init schema");
    prewarm(&database.pool).await;
    Some(database)
}

async fn memory_authority_tenants_are_isolated_and_foreign_ids_reveal_nothing() {
    let f = fixture().await;
    let op = f.operator.clone();
    f.tenant(&op, "tenant-a", &["shared-name"]).await;
    f.tenant(&op, "tenant-b", &["shared-name"]).await;
    let a = f
        .client(&op, "tenant-a", json!([grant("shared-name", 9, Some(9))]))
        .await
        .unwrap();
    let b = f
        .client(&op, "tenant-b", json!([grant("shared-name", 9, Some(9))]))
        .await
        .unwrap();
    let (Principal::Client(a), Principal::Client(b)) =
        (f.login(&a).await.unwrap(), f.login(&b).await.unwrap())
    else {
        panic!("clients")
    };
    assert_ne!(a.tenant_id, b.tenant_id);
    assert_ne!(a.grants[0].namespace_id, b.grants[0].namespace_id);
    // Granting a namespace of another tenant is impossible: names resolve within the tenant.
    assert_eq!(
        f.client(&op, "tenant-a", json!([grant("only-in-b", 1, None)]))
            .await
            .unwrap_err(),
        ErrorCode::InvalidRequest
    );
    // A foreign credential id is indistinguishable from a missing one.
    assert_eq!(
        f.run(
            &op,
            "admin.credentials.revoke",
            json!({"tenant": "tenant-a", "credential_id": b.credential_id})
        )
        .await
        .unwrap_err(),
        ErrorCode::NotFound
    );
    // The credential endpoint lists only the caller's own grants.
    let view = f
        .run(&Principal::Client(a.clone()), "credential.get", json!({}))
        .await
        .unwrap();
    assert_eq!(view["tenant"], "tenant-a");
    assert_eq!(view["grants"].as_array().unwrap().len(), 1);
}

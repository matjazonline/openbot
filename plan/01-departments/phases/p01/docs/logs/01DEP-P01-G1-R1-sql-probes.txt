## Probe A: shape rule reconstruction (D2) on a minimal table
```sql
\set ON_ERROR_STOP 0
DROP SCHEMA IF EXISTS s CASCADE; CREATE SCHEMA s; SET search_path=s,public;
CREATE TABLE departments(id int PRIMARY KEY, company_id int NOT NULL, parent_id int, UNIQUE(company_id,id),
  FOREIGN KEY(company_id,parent_id) REFERENCES departments(company_id,id),
  CONSTRAINT departments_not_own_parent CHECK (parent_id IS NULL OR parent_id <> id));
-- Reconstruction of DEP-P01-D2 as written: cycle if proposed_parent = checked or upward walk from
-- proposed_parent reaches checked; depth if ancestor_depth(proposed_parent) + subtree_height(checked) > 8.
CREATE FUNCTION department_shape_problem(c int, d int, p int) RETURNS text LANGUAGE sql STABLE AS $$
WITH RECURSIVE up(id, parent_id, n) AS (
    SELECT id, parent_id, 1 FROM s.departments WHERE company_id=c AND id=p
  UNION ALL
    SELECT x.id, x.parent_id, up.n+1 FROM s.departments x JOIN up ON x.id=up.parent_id AND x.company_id=c
     WHERE up.n < 9 AND up.id <> d),
down(id, n) AS (
    SELECT d, 1
  UNION ALL
    SELECT x.id, down.n+1 FROM s.departments x JOIN down ON x.parent_id=down.id AND x.company_id=c
     WHERE down.n < 9)
SELECT CASE
  WHEN p IS NULL THEN CASE WHEN (SELECT max(n) FROM down) > 8 THEN 'depth' END
  WHEN p = d OR EXISTS (SELECT 1 FROM up WHERE id=d) THEN 'cycle'
  WHEN coalesce((SELECT max(n) FROM up),0) + (SELECT max(n) FROM down) > 8 THEN 'depth'
END $$;
CREATE FUNCTION guard_department_shape() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE problem text := department_shape_problem(NEW.company_id, NEW.id, NEW.parent_id);
BEGIN
  IF problem IS NOT NULL THEN
    RAISE EXCEPTION 'department shape: %', problem USING ERRCODE='23514', CONSTRAINT='departments_shape';
  END IF; RETURN NEW;
END $$;
CREATE TRIGGER guard_department_shape BEFORE INSERT OR UPDATE OF parent_id ON departments FOR EACH ROW EXECUTE FUNCTION guard_department_shape();
-- chain 1..8 (level 1..8)
INSERT INTO departments VALUES (1,1,NULL);
INSERT INTO departments SELECT g,1,g-1 FROM generate_series(2,8) g;
\echo '--- 9th level insert (expect depth)'
INSERT INTO departments VALUES (9,1,8);
\echo '--- separate tree: 20 -> 21 -> 22 (height 3)'
INSERT INTO departments VALUES (20,1,NULL),(21,1,20),(22,1,21);
\echo '--- re-parent 20 under 5 (level 5 + height 3 = 8, expect ok)'
UPDATE departments SET parent_id=5 WHERE id=20;
\echo '--- re-parent 20 under 6 (6+3=9, expect depth)'
UPDATE departments SET parent_id=6 WHERE id=20;
\echo '--- cycle: re-parent 1 under 22 (22 is in 1 subtree now; expect cycle)'
UPDATE departments SET parent_id=22 WHERE id=1;
\echo '--- self parent (expect not_own_parent check or cycle)'
UPDATE departments SET parent_id=3 WHERE id=3;
\echo '--- move 20 back to root (expect ok)'
UPDATE departments SET parent_id=NULL WHERE id=20;
\echo '--- sibling move within subtree: 22 under 20 (already) and 21 under 22?? (cycle)'
UPDATE departments SET parent_id=22 WHERE id=21;
\echo '--- UPDATE not touching parent_id bypasses trigger (fine)'
\echo '--- multi-row UPDATE forming a cycle in ONE statement: 30->31, swap parents'
INSERT INTO departments VALUES (30,1,NULL),(31,1,NULL);
UPDATE departments SET parent_id = CASE id WHEN 30 THEN 31 WHEN 31 THEN 30 END WHERE id IN (30,31);
SELECT id,parent_id FROM departments WHERE id IN (30,31) ORDER BY id;
```

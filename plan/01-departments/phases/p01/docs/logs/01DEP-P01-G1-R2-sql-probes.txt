```sql
\set ON_ERROR_STOP 1
BEGIN;
CREATE TABLE public.departments (
  id uuid PRIMARY KEY, company_id uuid NOT NULL REFERENCES public.companies(id) ON DELETE CASCADE,
  name text NOT NULL, slug public.citext NOT NULL, parent_id uuid,
  created_by jsonb NOT NULL, created_at timestamptz NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT departments_not_own_parent CHECK (parent_id IS NULL OR parent_id <> id),
  UNIQUE (company_id, slug), UNIQUE (company_id, id));
ALTER TABLE public.departments ADD CONSTRAINT departments_parent_fk FOREIGN KEY (company_id, parent_id) REFERENCES public.departments(company_id, id);
CREATE TABLE public.department_members (company_id uuid NOT NULL, department_id uuid NOT NULL, principal_id uuid NOT NULL,
  created_at timestamptz NOT NULL DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (department_id, principal_id),
  FOREIGN KEY (company_id, department_id) REFERENCES public.departments(company_id, id) ON DELETE CASCADE,
  FOREIGN KEY (company_id, principal_id) REFERENCES public.principals(company_id, id) ON DELETE CASCADE);
CREATE TABLE public.department_roles (company_id uuid NOT NULL, department_id uuid NOT NULL, role text NOT NULL, principal_id uuid NOT NULL,
  created_at timestamptz NOT NULL DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (department_id, role, principal_id),
  CONSTRAINT department_roles_role_check CHECK (role = 'lead'),
  FOREIGN KEY (company_id, department_id) REFERENCES public.departments(company_id, id) ON DELETE CASCADE,
  FOREIGN KEY (company_id, principal_id) REFERENCES public.principals(company_id, id) ON DELETE CASCADE);
CREATE UNIQUE INDEX department_roles_one_lead ON public.department_roles (department_id, role) WHERE role = 'lead';
-- r2 D2 kind guard (reconstruction): reads principal FOR KEY SHARE
CREATE FUNCTION public.guard_department_principal_kind() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE k text;
BEGIN
  SELECT kind INTO k FROM public.principals WHERE company_id=NEW.company_id AND id=NEW.principal_id FOR KEY SHARE;
  IF k IS NULL OR k NOT IN ('person','agent') THEN
    RAISE EXCEPTION 'department principal must be a person or agent' USING ERRCODE='23514', CONSTRAINT='department_principal_kind';
  END IF;
  RETURN NEW;
END $$;
CREATE TRIGGER guard_department_principal_kind BEFORE INSERT OR UPDATE ON public.department_members FOR EACH ROW EXECUTE FUNCTION public.guard_department_principal_kind();
CREATE TRIGGER guard_department_principal_kind BEFORE INSERT OR UPDATE ON public.department_roles FOR EACH ROW EXECUTE FUNCTION public.guard_department_principal_kind();
-- r2 D1 demotion trigger (reconstruction)
CREATE FUNCTION public.department_principal_demoted() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  DELETE FROM public.department_members WHERE company_id=NEW.company_id AND principal_id=NEW.id;
  DELETE FROM public.department_roles WHERE company_id=NEW.company_id AND principal_id=NEW.id;
  RETURN NULL;
END $$;
CREATE TRIGGER department_principal_demoted AFTER UPDATE OF kind ON public.principals FOR EACH ROW
  WHEN (NEW.kind NOT IN ('person','agent')) EXECUTE FUNCTION public.department_principal_demoted();
COMMIT;
```

import os,re,sys
root=sys.argv[1]
word=re.compile(r'(?i)\bchannels?\b')
def literals(src):
    i=0;n=len(src);out=[]
    while i<n:
        c=src[i]
        if src.startswith('//',i):
            j=src.find('\n',i); i=n if j<0 else j; continue
        if src.startswith('/*',i):
            j=src.find('*/',i); i=n if j<0 else j+2; continue
        m=re.match(r'b?r(#*)"',src[i:])
        if m and (i==0 or not (src[i-1].isalnum() or src[i-1]=='_')):
            hashes=m.group(1); start=i+m.end(); end=src.find('"'+hashes,start)
            out.append((src.count('\n',0,i)+1,src[start:end])); i=end+1+len(hashes); continue
        if c=="'" :
            # char literal or lifetime
            m2=re.match(r"'(\\.|[^\\'])'",src[i:])
            i+= m2.end() if m2 else 1; continue
        if c=='"':
            j=i+1;buf=[]
            while j<n and src[j]!='"':
                if src[j]=='\\': buf.append(src[j:j+2]); j+=2; continue
                buf.append(src[j]); j+=1
            out.append((src.count('\n',0,i)+1,''.join(buf))); i=j+1; continue
        i+=1
    return out
for dp,_,fs in os.walk(root):
    for f in fs:
        if not f.endswith('.rs') or f=='tests.rs' or f.endswith('_tests.rs'): continue
        p=os.path.join(dp,f); s=open(p,encoding='utf-8').read()
        for line,lit in literals(s):
            if '<' in lit or '>' in lit: continue  # markup literals are covered by the author's scan
            if word.search(lit):
                print(f"{p}:{line}: {lit[:140]!r}")

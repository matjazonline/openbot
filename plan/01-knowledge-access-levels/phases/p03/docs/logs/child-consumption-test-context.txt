async fn workflow_child_timeout_fails_child_and_reaches_parent_error_route() {
    let (f, parent) = AdmissionFixture::with_child(round_child(), |child, version| {
        let mut source = routed(child, version);
        source["steps"]["start"]["with"]["input"] =
            json!({"literal": {"previous_body": "", "feedback": ""}});
        source
    })
    .await;
    let p = f.persistence();
    let child = f.spawn(parent).await;
    p.park_wait(child).await.unwrap().unwrap();
    // The child's own wait passes its deadline; its run fails and the parent is told.
    sqlx::query("UPDATE workflow_waits SET created_at=clock_timestamp()-interval '2 seconds',deadline=clock_timestamp()-interval '1 second' WHERE run_id=$1")
        .bind(child.run.as_uuid()).execute(p.pool()).await.unwrap();
    assert_eq!(p.sweep_waits(8).await.unwrap(), 1);
    assert_eq!(state(&f, child.run).await, "failed");
    let WaitProgress::Completed(step) = p.resume_wait(parent).await.unwrap() else {
        panic!("routed")
    };
    assert_eq!(step.route, RouteSelection::FinalError);
    // The child's deadline is the parent's: it can never outlive it.
    let deadlines: (chrono::DateTime<chrono::Utc>, chrono::DateTime<chrono::Utc>) = sqlx::query_as(
        "SELECT parent.deadline, child.deadline FROM workflow_runs AS parent JOIN workflow_runs AS child ON child.parent_run_id=parent.id WHERE parent.id=$1",
    )
    .bind(parent.run.as_uuid())
    .fetch_one(p.pool())
    .await
    .unwrap();
    assert_eq!(deadlines.0, deadlines.1);
}

async fn workflow_tool_child_output_outside_the_pinned_schema_fails_the_parent_wait() {
    let setup = Box::pin(ToolSetup::new(echo_child())).await;
    let p = setup.persistence();
    let claim = setup.claim("invalid-output").await;
    let scope = claim.fence.scope;
    let call = setup.call(
        setup.frozen_pin(setup.parent).await,
        json!({"question": "q"}),
    );
    setup.dispatch(&claim, &call).await.unwrap();
    let child = setup.finish_child(scope).await;
    // Corrupted outside the runtime (triggers bypassed for this one raw update), as only a
    // defect or a manual edit could; the parent re-checks it against its pinned copy.
    let mut tx = p.pool().begin().await.unwrap();
    sqlx::query("SET LOCAL session_replication_role = replica")
        .execute(&mut *tx)
        .await
        .unwrap();
    let corrupted = sqlx::query("UPDATE workflow_executions SET committed_output='{\"choice\":\"maybe\",\"answer\":\"q\"}'::jsonb WHERE run_id=$1 AND completed_at IS NOT NULL")
        .bind(child.run.as_uuid())
        .execute(&mut *tx)
        .await
        .unwrap();
    assert_eq!(corrupted.rows_affected(), 1);
    tx.commit().await.unwrap();
    for disposition in [CommitDisposition::Committed, CommitDisposition::Replayed] {
        assert_eq!(
            p.resume_wait(scope).await.unwrap(),
            WaitProgress::Failed {
                disposition,
                reason: WaitFailureReason::InvalidOutput,
            }
        );
    }
    let settled: (String, String, Option<String>) = sqlx::query_as(
        "SELECT run.state, wait.state, wait.failure_code FROM workflow_runs AS run \
         JOIN workflow_waits AS wait ON wait.company_id=run.company_id AND wait.run_id=run.id \
         WHERE run.id=$1 AND wait.reason='child_run'",
    )
    .bind(scope.run.as_uuid())
    .fetch_one(p.pool())
    .await
    .unwrap();
    assert_eq!(
        settled,
        (
            "failed".into(),
            "failed".into(),
            Some("invalid_output".into())
        )
    );
}

    for filter in [
        DEFAULT_LOG_FILTER,
        configured_filter(include_str!("../../fly.toml")),

    ] {
        let output = Arc::new(Mutex::new(Vec::new()));
        let sink = output.clone();
        let subscriber = tracing_subscriber::registry()
            .with(EnvFilter::new(filter))
            .with(
                fmt::layer()
                    .without_time()
                    .with_ansi(false)
                    .with_target(false)
                    .with_writer(move || Capture(sink.clone())),
            );
        tracing::subscriber::with_default(subscriber, || {
            let monitor = TracingMonitor::new();
            monitor.increment_counter(
                "session_validation_failures",
                1,
                &[("reason", "store_unavailable")],
            );
            monitor.record_gauge("session_cleanup_backlog", 1001.0, &[]);
            tracing::info!(target: "open_bots::session_probe", "application-marker");
            tracing::info!(target: "monitoring::smtp", "excluded-smtp-marker");
            tracing::info!(target: "monitoring::ai", "excluded-ai-marker");
        });
        let captured = String::from_utf8(output.lock().unwrap().clone()).unwrap();

            captured.contains("application-marker"),
            "{filter}: {captured}"
        );
        assert!(
            captured.contains("session_validation_failures"),
            "{filter}: {captured}"
        );
        assert!(
            captured.contains("store_unavailable"),
            "{filter}: {captured}"
        );
        assert!(
            captured.contains("session_cleanup_backlog"),
            "{filter}: {captured}"
        );
        assert!(!captured.contains("excluded-smtp-marker"));

    async fn read_credential(&self, scope: &CredentialScope) -> AppResult<Option<SecretString>> {
        let cipher = self.credential_cipher()?;
        let envelope: Option<String> = sqlx::query_scalar(
            r#"SELECT credential.envelope
               FROM integration_credentials AS credential
               JOIN integration_installations AS installation
                   ON installation.company_id = credential.company_id
                  AND installation.id = credential.installation_id
               WHERE credential.company_id = $1
                 AND credential.installation_id = $2
                 AND credential.credential_kind = $3
                 AND installation.transport = $4"#,
        )
        .bind(scope.company_id)
        .bind(scope.installation_id.as_uuid())
        .bind(scope.kind.as_str())
        .bind(scope.transport.as_str())
        .fetch_optional(&self.pool)
        .await
        .map_err(AppError::from)?;

        let Some(envelope) = envelope else {
            return Ok(None);
        };
        cipher
            .open_envelope(&context(scope), &envelope)
            .map(Some)
            .inspect_err(|error| {
                // Identifiers and the failure class only; `EnvelopeError` is written so that its
                // own text carries neither ciphertext nor key material.
                warn!(
                    company_id = %scope.company_id,
                    installation_id = %scope.installation_id,
                    credential_kind = scope.kind.as_str(),
                    failure = %error,
                    "stored integration credential failed to open"
                );
            })
    }
